def feature_amount_interval_ratio(open_df, tx_df, close_df):
    """
    金额-间隔比率特征：计算每个钱包的金额与交易间隔的交互特征
    基于领域知识：
    - 小额测试交易(small_amount_testing)：金额极小但交易频繁，间隔短
    - 在线洗钱(online_laundering)：交易频繁，金额中等偏低
    - 四类钱包洗钱(class4_laundering)：高频快速流转
    这些风险模式表现为金额与间隔的异常比率组合
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['amount_abs'] = tx['amount'].abs()
    
    # 计算每个钱包的金额-间隔交互特征
    src_stats = tx.groupby('src').agg(
        tx_cnt=('tx_id', 'count'),
        amt_mean=('amount_abs', 'mean'),
        amt_median=('amount_abs', 'median'),
        amt_std=('amount_abs', 'std'),
        interval_mean=('interval', 'mean'),
        interval_median=('interval', 'median'),
        interval_std=('interval', 'std'),
        # 极短间隔交易占比（间隔<60秒）
        short_interval_count=('interval', lambda x: (x < 60).sum()),
        # 极小金额交易占比（金额<=100）
        tiny_amount_count=('amount_abs', lambda x: (x <= 100).sum()),
    ).reset_index()
    
    # 金额与间隔的比率特征
    src_stats['amt_interval_ratio'] = src_stats['amt_mean'] / src_stats['interval_mean'].clip(lower=1)
    src_stats['amt_interval_median_ratio'] = src_stats['amt_median'] / src_stats['interval_median'].clip(lower=1)
    
    # 短间隔小额交易占比（测试交易的特征）
    src_stats['short_interval_ratio'] = src_stats['short_interval_count'] / src_stats['tx_cnt'].clip(lower=1)
    src_stats['tiny_amount_ratio'] = src_stats['tiny_amount_count'] / src_stats['tx_cnt'].clip(lower=1)
    
    # 金额与间隔的交互得分：高频率+小金额 = 洗钱特征
    src_stats['freq_small_score'] = src_stats['short_interval_ratio'] * src_stats['tiny_amount_ratio']
    
    # 金额CV与间隔CV的比值 - 衡量金额和时间的协同波动
    src_stats['amt_cv'] = src_stats['amt_std'] / src_stats['amt_mean'].clip(lower=0.01)
    src_stats['interval_cv'] = src_stats['interval_std'] / src_stats['interval_mean'].clip(lower=0.01)
    src_stats['amt_interval_cv_ratio'] = src_stats['amt_cv'] / src_stats['interval_cv'].clip(lower=0.01)
    
    result = src_stats[['src', 'amt_interval_ratio', 'freq_small_score', 'amt_interval_cv_ratio']].copy()
    result.columns = ['wallet_id', 'amt_interval_ratio', 'freq_small_score', 'amt_interval_cv_ratio']
    
    # 处理无穷值和NaN
    for col in ['amt_interval_ratio', 'freq_small_score', 'amt_interval_cv_ratio']:
        result[col] = result[col].replace([np.inf, -np.inf], 0).fillna(0)
    
    return result

def feature_amount_sequential_pattern(open_df, tx_df, close_df):
    """
    金额序列模式特征：分析每个钱包交易金额的时序变化模式
    基于领域知识：
    - 小额测试交易(small_amount_testing)：先小额测试后大额
    - 洗钱交易金额呈现"先小后大"或"均匀分散"的模式
    - 正常交易金额有自然波动，但不会出现明显的测试-大额模式
    
    通过分析金额序列的自相关性、变化方向和模式来捕捉异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['amount_abs'] = tx['amount'].abs()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按src和时间排序
    tx = tx.sort_values(['src', 'timestamp']).reset_index(drop=True)
    
    # 计算每个钱包的金额序列特征
    def calc_seq_features(group):
        amounts = group['amount_abs'].values
        n = len(amounts)
        
        if n < 2:
            return pd.Series({
                'amt_autocorr': 0,
                'amt_trend': 0,
                'amt_first_last_ratio': 0,
                'amt_rising_ratio': 0,
                'amt_zigzag_count': 0
            })
        
        # 1. 金额自相关（lag-1）- 衡量金额的持续性
        mean_amt = np.mean(amounts)
        std_amt = np.std(amounts)
        if std_amt > 0:
            amt_norm = (amounts - mean_amt) / std_amt
            autocorr = np.mean(amt_norm[:-1] * amt_norm[1:])
        else:
            autocorr = 0
        
        # 2. 金额趋势 - 线性回归斜率
        x = np.arange(n)
        if n > 2:
            slope = np.polyfit(x, amounts, 1)[0]
        else:
            slope = amounts[-1] - amounts[0] if n == 2 else 0
        
        # 3. 首笔/末笔金额比
        first_last_ratio = amounts[-1] / amounts[0] if amounts[0] > 0 else 1
        
        # 4. 上涨交易占比（金额递增的比例）
        diffs = np.diff(amounts)
        rising_ratio = np.sum(diffs > 0) / len(diffs) if len(diffs) > 0 else 0
        
        # 5. 锯齿次数（方向变化的次数）
        if len(diffs) > 1:
            sign_changes = np.sum(np.diff(np.sign(diffs)) != 0)
        else:
            sign_changes = 0
        
        return pd.Series({
            'amt_autocorr': autocorr,
            'amt_trend': slope,
            'amt_first_last_ratio': min(first_last_ratio, 100),  # 截断异常值
            'amt_rising_ratio': rising_ratio,
            'amt_zigzag_count': sign_changes / max(n-1, 1)  # 归一化
        })
    
    seq_features = tx.groupby('src', group_keys=False).apply(calc_seq_features).reset_index()
    
    # 综合异常模式得分
    # 低自相关+强趋势+高首末比 = 测试交易模式
    # 高锯齿+低趋势 = 分散转出模式
    seq_features['amount_seq_anomaly_score'] = (
        (1 - np.abs(seq_features['amt_autocorr'].clip(-1, 1))) * 0.3 +
        np.abs(seq_features['amt_trend']) / (seq_features['amt_trend'].abs().max() + 1e-5) * 0.3 +
        seq_features['amt_zigzag_count'] * 0.4
    )
    
    result = seq_features[['src', 'amount_seq_anomaly_score', 'amt_autocorr', 'amt_trend']].copy()
    result.columns = ['wallet_id', 'amount_seq_anomaly_score', 'amt_autocorr', 'amt_trend']
    
    # 处理NaN
    for col in ['amount_seq_anomaly_score', 'amt_autocorr', 'amt_trend']:
        result[col] = result[col].fillna(0)
    
    return result

def feature_hourly_entropy_workdev(open_df, tx_df, close_df):
    """
    小时熵值与工作时段偏离度：衡量交易时间的分散程度和对工作时段的偏离
    基于领域知识：
    - 洗钱交易集中在特定时段（凌晨/深夜），时间分布不均匀
    - 正常交易全天分布更均匀，符合自然生活规律
    - 异常钱包工作时段占比更低(66.2% vs 71.9%)
    
    通过小时熵值（衡量分散度）和工作时段偏离度来识别异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    open_tmp = open_df[['wallet_id', 'is_abnormal']].rename(columns={'wallet_id':'src'})
    
    # 1. 小时熵值 - 每个钱包的交易时间分散度
    def calc_hour_entropy(group):
        hour_counts = group['hour'].value_counts()
        probs = hour_counts / hour_counts.sum()
        return -np.sum(probs * np.log2(probs + 1e-10))
    
    entropy_vals = tx.groupby('src').apply(calc_hour_entropy).reset_index()
    entropy_vals.columns = ['src', 'hour_entropy']
    
    # 2. 工作时段(9-17)交易占比
    tx['is_work_hour'] = ((tx['hour'] >= 9) & (tx['hour'] <= 17)).astype(int)
    work_ratio = tx.groupby('src')['is_work_hour'].mean().reset_index()
    work_ratio.columns = ['src', 'work_hour_ratio']
    
    # 3. 工作时段偏离度 - 与正常均值(71.9%)的偏离
    # 正常钱包工作时段占比约71.9%，异常约66.2%
    work_ratio['work_deviation'] = np.abs(work_ratio['work_hour_ratio'] - 0.719)
    
    # 4. 凌晨(0-5)交易占比
    tx['is_dawn'] = (tx['hour'] <= 5).astype(int)
    dawn_ratio = tx.groupby('src')['is_dawn'].mean().reset_index()
    dawn_ratio.columns = ['src', 'dawn_ratio']
    
    # 5. 深夜(22-23)交易占比
    tx['is_late'] = (tx['hour'] >= 22).astype(int)
    late_ratio = tx.groupby('src')['is_late'].mean().reset_index()
    late_ratio.columns = ['src', 'late_ratio']
    
    # 合并
    result = entropy_vals.merge(work_ratio, on='src')
    result = result.merge(dawn_ratio, on='src')
    result = result.merge(late_ratio, on='src')
    
    # 综合得分
    # 低熵值（时间集中）+ 高工作时段偏离 + 高凌晨/深夜占比 = 异常
    max_entropy = np.log2(24)  # 最大可能熵值
    result['time_entropy_norm'] = result['hour_entropy'] / max_entropy
    
    result['hourly_anomaly_score'] = (
        (1 - result['time_entropy_norm']) * 0.3 +  # 低熵=异常
        result['work_deviation'] * 0.3 +  # 偏离工作时段=异常
        result['dawn_ratio'] * 2.0 +  # 凌晨交易权重高
        result['late_ratio'] * 1.5  # 深夜交易
    )
    
    result = result[['src', 'hourly_anomaly_score', 'hour_entropy', 'work_hour_ratio']].copy()
    result.columns = ['wallet_id', 'hourly_anomaly_score', 'hour_entropy', 'work_hour_ratio']
    
    return result

def feature_amount_dispersion_entropy(open_df, tx_df, close_df):
    """
    金额离散熵特征：衡量每个钱包交易金额的分散程度和多样性
    基于领域知识：
    - 洗钱交易金额分散度更高（拆分转出、多笔小额）
    - 正常交易金额分布更集中（工资、消费等有明确金额）
    - 异常交易的金额CV(1.21)高于正常(1.04)，说明金额更分散
    
    使用熵值衡量金额的多样性，结合离散系数衡量波动性
    """
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    tx['amount_abs'] = tx['amount'].abs()
    
    # 1. 金额离散度 - 使用IQR/中位数比
    src_stats = tx.groupby('src').agg(
        tx_cnt=('tx_id', 'count'),
        amt_mean=('amount_abs', 'mean'),
        amt_median=('amount_abs', 'median'),
        amt_std=('amount_abs', 'std'),
        amt_max=('amount_abs', 'max'),
        amt_min=('amount_abs', 'min'),
        amt_q25=('amount_abs', lambda x: x.quantile(0.25)),
        amt_q75=('amount_abs', lambda x: x.quantile(0.75)),
    ).reset_index()
    
    # 离散度指标
    src_stats['amt_cv'] = src_stats['amt_std'] / src_stats['amt_mean'].clip(lower=0.01)
    src_stats['amt_iqr'] = src_stats['amt_q75'] - src_stats['amt_q25']
    src_stats['amt_iqr_median_ratio'] = src_stats['amt_iqr'] / src_stats['amt_median'].clip(lower=0.01)
    src_stats['amt_range_median_ratio'] = (src_stats['amt_max'] - src_stats['amt_min']) / src_stats['amt_median'].clip(lower=0.01)
    
    # 2. 金额熵值 - 衡量金额多样性
    # 对金额进行分桶（按100元为粒度）
    tx['amount_bin'] = (tx['amount_abs'] // 100).astype(int)
    
    def calc_entropy(group):
        counts = group['amount_bin'].value_counts()
        probs = counts / counts.sum()
        return -np.sum(probs * np.log2(probs + 1e-10))
    
    entropy_vals = tx.groupby('src').apply(calc_entropy).reset_index()
    entropy_vals.columns = ['src', 'amount_entropy']
    
    # 3. 金额集中度 - top3金额占比
    def top3_concentration(group):
        top3 = group['amount_abs'].value_counts().nlargest(3).sum()
        total = len(group)
        return top3 / total if total > 0 else 0
    
    concentration = tx.groupby('src').apply(top3_concentration).reset_index()
    concentration.columns = ['src', 'amount_top3_concentration']
    
    # 合并
    result = src_stats.merge(entropy_vals, on='src').merge(concentration, on='src')
    
    # 综合离散得分
    result['amount_dispersion_score'] = (
        result['amt_cv'] * 0.3 +
        result['amt_iqr_median_ratio'] * 0.2 +
        result['amount_entropy'] * 0.3 +
        (1 - result['amount_top3_concentration']) * 0.2
    )
    
    result = result[['src', 'amount_dispersion_score', 'amount_entropy', 'amount_top3_concentration']].copy()
    result.columns = ['wallet_id', 'amount_dispersion_score', 'amount_entropy', 'amount_top3_concentration']
    
    # 处理NaN
    for col in ['amount_dispersion_score', 'amount_entropy', 'amount_top3_concentration']:
        result[col] = result[col].fillna(0)
    
    return result

def feature_entity_device_concentration(open_df, tx_df, close_df):
    """
    设备集中度特征：衡量每个钱包使用的设备集中程度
    基于领域知识：
    - 异常钱包使用更少的设备(1.62 vs 4.18)，由单一操作员控制
    - 正常钱包使用多个设备（手机、电脑等不同环境）

    通过设备数/交易笔数的比率和设备集中度来识别异常
    """

    import numpy as np
    
    tx = tx_df.copy()
    
    # 1. 源设备统计
    src_device = tx.groupby('src').agg(
        src_device_count=('src_device', 'nunique'),
        src_tx_cnt=('tx_id', 'count'),
        src_device_most_used=('src_device', lambda x: x.value_counts().max())
    ).reset_index()
    
    # 设备集中度 - 最常用设备的使用占比
    src_device['src_device_concentration'] = src_device['src_device_most_used'] / src_device['src_tx_cnt'].clip(lower=1)
    # 设备数/交易笔数比 - 低值表示设备集中
    src_device['src_device_per_tx'] = src_device['src_device_count'] / src_device['src_tx_cnt'].clip(lower=1)

    dst_device = tx.groupby('dst').agg(
        dst_device_count=('dst_device', 'nunique'),
        dst_tx_cnt=('tx_id', 'count'),
        dst_device_most_used=('dst_device', lambda x: x.value_counts().max())
    ).reset_index()
    dst_device.columns = ['wallet_id', 'dst_device_count', 'dst_tx_cnt', 'dst_device_most_used']
    dst_device['dst_device_concentration'] = dst_device['dst_device_most_used'] / dst_device['dst_tx_cnt'].clip(lower=1)
    dst_device['dst_device_per_tx'] = dst_device['dst_device_count'] / dst_device['dst_tx_cnt'].clip(lower=1)
    
    # 3. 源设备和目标设备的共享情况
    # 每个设备被多少钱包使用

    device_wallet_count.columns = ['src_device', 'device_wallet_share']

    result.columns = ['wallet_id', 'src_device_concentration', 'src_device_count', 'src_device_per_tx']
    
    dst_result = dst_device[['wallet_id', 'dst_device_concentration', 'dst_device_count', 'dst_device_per_tx']]
    result = result.merge(dst_result, on='wallet_id', how='outer').fillna(0)
    result = result.merge(device_share, left_on='wallet_id', right_on='src', how='left').drop('src', axis=1)
    result['avg_device_share'] = result['avg_device_share'].fillna(0)
    
    # 综合设备异常得分
    # 高设备集中度（少设备）+ 低设备共享 = 异常（单一操作员）
    result['device_anomaly_score'] = (
        result['src_device_concentration'] * 0.3 +  # 设备集中度高=异常
        (1 - result['src_device_per_tx'].clip(upper=1)) * 0.3 +  # 每笔交易设备数少=异常
        result['dst_device_concentration'] * 0.2 +
        (1 - result['avg_device_share'].clip(upper=1)) * 0.2
    )
    
    result = result[['wallet_id', 'device_anomaly_score', 'src_device_concentration', 'src_device_count']].copy()
    
    return result

def feature_time_hourly_burst_pattern(open_df, tx_df, close_df):
    """
    小时级突发交易模式特征：衡量交易在小时内的集中爆发程度
    基于领域知识：
    - 洗钱交易呈现"突发模式"：短时间内连续多笔交易后长时间静默
    - 正常交易分布更均匀
    - 异常钱包的瞬时交易(<=10s)占比(29.6%)远低于正常(61.9%)
    
    通过每小时的交易集中度（Gini系数）和突发指数来识别异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按小时聚合交易笔数，使用 'h' 而不是 'H'
    tx['hour_bin'] = tx['timestamp'].dt.floor('h')
    
    # 每个钱包的每小时交易笔数分布
    hourly_counts = tx.groupby(['src', 'hour_bin']).size().reset_index(name='hourly_tx_cnt')
    
    # 计算每个钱包的Gini系数（衡量小时交易量的不均匀度）
    def calc_gini(group):
        values = group['hourly_tx_cnt'].values
        if len(values) < 2:
            return 0
        values = np.sort(values)
        n = len(values)
        index = np.arange(1, n + 1)
        return (2 * np.sum(index * values) / (n * np.sum(values)) - (n + 1) / n)
    
    gini = hourly_counts.groupby('src').apply(calc_gini).reset_index()
    gini.columns = ['src', 'hourly_gini']
    
    # 最大小时交易量占比
    max_hourly = hourly_counts.groupby('src')['hourly_tx_cnt'].max().reset_index()
    max_hourly.columns = ['src', 'max_hourly_tx']
    total_tx = tx.groupby('src').size().reset_index(name='total_tx')
    max_hourly = max_hourly.merge(total_tx, on='src')
    max_hourly['max_hour_ratio'] = max_hourly['max_hourly_tx'] / max_hourly['total_tx'].clip(lower=1)
    
    # 合并
    result = gini.merge(max_hourly[['src', 'max_hour_ratio']], on='src')
    
    # 突发得分：高Gini（分布不均）+ 高最大小时占比 = 突发模式
    result['burst_score'] = (
        result['hourly_gini'] * 0.5 +
        result['max_hour_ratio'] * 0.5
    )
    
    result = result[['src', 'burst_score', 'hourly_gini', 'max_hour_ratio']].copy()
    result.columns = ['wallet_id', 'burst_score', 'hourly_gini', 'max_hour_ratio']
    
    return result

def feature_behavior_counterparty_overlap_decay(open_df, tx_df, close_df):
    """
    交易对手重叠衰减特征：衡量钱包前后期交易对手的变化程度
    基于领域知识：
    - 异常钱包对手前后重叠度(0.028)远低于正常(0.179)
    - 洗钱交易频繁更换交易对手以规避追踪
    - 正常交易有固定交易对手（亲友、商户）
    
    通过将交易序列分为前后两半，计算对手集合的Jaccard相似度
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx = tx.sort_values(['src', 'timestamp']).reset_index(drop=True)
    
    def calc_overlap_decay(group):
        n = len(group)
        if n < 4:
            return 0, 0, 0
        
        # 前1/3 vs 后1/3
        third = n // 3
        first = set(group['dst'].iloc[:third])
        last = set(group['dst'].iloc[-third:])
        
        overlap = 0
        if len(first) > 0 and len(last) > 0:
            overlap = len(first & last) / len(first | last)
        
        # 独特对手数
        unique_cp = group['dst'].nunique()
        
        # 对手更换率（相邻交易对手不同的比例）
        diffs = (group['dst'].values[1:] != group['dst'].values[:-1])
        change_rate = diffs.mean()
        
        return overlap, unique_cp, change_rate
    
    features = tx.groupby('src', group_keys=False).apply(
        lambda g: pd.Series(calc_overlap_decay(g), 
                           index=['cp_overlap_decay', 'unique_cp_count', 'cp_change_rate'])
    ).reset_index()
    
    # 综合得分：低重叠度 + 高独特对手数 + 高变更率 = 异常
    features['cp_decay_score'] = (
        (1 - features['cp_overlap_decay']) * 0.4 +
        (features['cp_change_rate']) * 0.3 +
        (features['unique_cp_count'] / features['unique_cp_count'].clip(upper=features['unique_cp_count'].max())) * 0.3
    )
    
    result = features[['src', 'cp_decay_score', 'cp_overlap_decay', 'cp_change_rate']].copy()
    result.columns = ['wallet_id', 'cp_decay_score', 'cp_overlap_decay', 'cp_change_rate']
    
    return result

def feature_behavior_direction_sequence_pattern(open_df, tx_df, close_df):
    """
    交易方向序列模式特征：分析钱包的发送/接收方向变化模式
    基于领域知识：
    - 洗钱交易呈现"接收→发送"的固定模式（收到资金后立即转出）
    - 正常交易的方向更随机（消费、转账等）
    - 异常钱包的发送/接收比(0.62)远低于正常(1.48)
    
    通过方向序列的熵值、转换模式和集中度来识别异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 构建每个钱包的完整交易序列（包括作为src和dst）
    # 作为发送方
    sent_tx = tx[['src', 'timestamp', 'amount']].copy()
    sent_tx['direction'] = 'send'
    sent_tx.columns = ['wallet_id', 'timestamp', 'amount', 'direction']
    
    # 作为接收方
    recv_tx = tx[['dst', 'timestamp', 'amount']].copy()
    recv_tx['direction'] = 'receive'
    recv_tx.columns = ['wallet_id', 'timestamp', 'amount', 'direction']
    
    # 合并并排序
    all_tx = pd.concat([sent_tx, recv_tx], ignore_index=True)
    all_tx = all_tx.sort_values(['wallet_id', 'timestamp']).reset_index(drop=True)
    
    # 计算方向序列特征
    def calc_direction_features(group):
        directions = group['direction'].values
        n = len(directions)
        
        if n < 2:
            return pd.Series({
                'send_ratio': 0,
                'direction_change_rate': 0,
                'direction_entropy': 0,
                'consecutive_send_max': 0,
                'consecutive_recv_max': 0
            })
        
        # 发送占比
        send_ratio = np.mean(directions == 'send')
        
        # 方向变更率
        changes = np.sum(directions[1:] != directions[:-1])
        direction_change_rate = changes / (n - 1)
        
        # 方向熵值
        send_p = send_ratio
        recv_p = 1 - send_p
        if send_p > 0 and recv_p > 0:
            direction_entropy = -send_p * np.log2(send_p) - recv_p * np.log2(recv_p)
        else:
            direction_entropy = 0
        
        # 最大连续发送/接收
        send_streaks = np.diff(np.where(np.concatenate(([True], directions[:-1] != 'send', [True])))[0])
        recv_streaks = np.diff(np.where(np.concatenate(([True], directions[:-1] != 'receive', [True])))[0])
        
        consecutive_send_max = np.max(send_streaks) if len(send_streaks) > 0 else 0
        consecutive_recv_max = np.max(recv_streaks) if len(recv_streaks) > 0 else 0
        
        return pd.Series({
            'send_ratio': send_ratio,
            'direction_change_rate': direction_change_rate,
            'direction_entropy': direction_entropy,
            'consecutive_send_max': consecutive_send_max,
            'consecutive_recv_max': consecutive_recv_max
        })
    
    # 使用采样加速（如果钱包太多）
    dir_features = all_tx.groupby('wallet_id', group_keys=False).apply(calc_direction_features).reset_index()
    
    # 综合方向序列异常得分
    # 正常：send_ratio≈0.60, 方向变更频繁, 熵值高
    # 异常：send_ratio≈0.38, 方向变更模式异常
    dir_features['direction_anomaly_score'] = (
        np.abs(dir_features['send_ratio'] - 0.60) * 0.3 +  # 偏离正常发送比
        (1 - dir_features['direction_entropy']) * 0.3 +  # 低熵=异常
        dir_features['consecutive_recv_max'] / (dir_features['consecutive_recv_max'].max() + 1) * 0.4  # 长连续接收=异常
    )
    
    result = dir_features[['wallet_id', 'direction_anomaly_score', 'send_ratio', 'direction_change_rate']].copy()
    
    return result

def feature_behavior_active_days_intensity(open_df, tx_df, close_df):
    """
    活跃天数与交易强度特征：衡量钱包的活跃周期和交易集中度
    基于领域知识：

    通过活跃天数、日均交易笔数和交易集中度来识别异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx['date'] = tx['timestamp'].dt.date
    
    # 1. 活跃天数

    active_days.columns = ['src', 'active_days']

    max_daily = daily_tx.groupby('src')['daily_cnt'].max().reset_index()

    daily_mean = daily_tx.groupby('src')['daily_cnt'].mean().reset_index()
    daily_mean.columns = ['src', 'daily_mean']
    
    daily_cv = daily_std.merge(daily_mean, on='src')
    daily_cv['daily_cv'] = daily_cv['daily_std'] / daily_cv['daily_mean'].clip(lower=0.01)
    
    # 合并
    result = intensity.merge(max_daily, on='src')
    result = result.merge(daily_cv[['src', 'daily_cv']], on='src')
    
    # 综合活跃强度得分
    # 异常：活跃天数少 + 日均交易多 + 日交易量波动大
    result['active_intensity_score'] = (
        (1 / result['active_days'].clip(lower=1)) * 0.3 +  # 活跃天数少=异常
        (result['tx_per_day'] / result['tx_per_day'].clip(upper=result['tx_per_day'].max())) * 0.3 +  # 日均交易多=异常
        (result['max_daily_tx'] / result['tx_cnt'].clip(lower=1)) * 0.4  # 最大日占比高=异常
    )
    
    result = result[['src', 'active_intensity_score', 'active_days', 'tx_per_day']].copy()
    result.columns = ['wallet_id', 'active_intensity_score', 'active_days', 'tx_per_day']
    
    return result

def feature_behavior_tx_timing_regularity(open_df, tx_df, close_df):
    """
    交易时间规律性特征：衡量每个钱包交易时间的规律程度
    基于领域知识：
    - 异常钱包的日均交易笔数(1.33)低于正常(1.62)
    - 异常钱包同日交易占比更低（交易分散在不同日期）
    - 洗钱交易的时间间隔更有规律（操作员按指令操作）
    - 正常交易的时间间隔更随机
    
    通过交易间隔的规律性（变异系数、自相关）来识别异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx = tx.sort_values(['src', 'timestamp']).reset_index(drop=True)
    
    # 计算每个钱包的交易间隔统计
    interval_stats = tx.groupby('src')['interval'].agg(
        interval_mean='mean',
        interval_median='median',
        interval_std='std',
        interval_min='min',
        interval_max='max',
        interval_q25=lambda x: x.quantile(0.25),
        interval_q75=lambda x: x.quantile(0.75),
    ).reset_index()
    
    # 间隔变异系数
    interval_stats['interval_cv'] = interval_stats['interval_std'] / interval_stats['interval_mean'].clip(lower=0.01)
    
    # 间隔的IQR/中位数比（衡量规律性）
    interval_stats['interval_iqr'] = interval_stats['interval_q75'] - interval_stats['interval_q25']
    interval_stats['interval_iqr_median_ratio'] = interval_stats['interval_iqr'] / interval_stats['interval_median'].clip(lower=0.01)
    
    # 间隔的偏度（(均值-中位数)/标准差）
    interval_stats['interval_skew'] = (interval_stats['interval_mean'] - interval_stats['interval_median']) / interval_stats['interval_std'].clip(lower=0.01)
    
    # 交易时间规律性得分
    # 规律性高（低CV + 低IQR/中位数比）= 可能有规律的洗钱操作
    # 规律性低（高CV + 高IQR/中位数比）= 正常随机交易
    interval_stats['timing_regularity_score'] = (
        (1 / (1 + interval_stats['interval_cv'])) * 0.4 +  # 低CV=规律=异常
        (1 / (1 + interval_stats['interval_iqr_median_ratio'])) * 0.3 +  # 低IQR比=规律=异常
        np.abs(interval_stats['interval_skew'].clip(-5, 5)) / 5 * 0.3  # 高偏度=异常
    )
    
    result = interval_stats[['src', 'timing_regularity_score', 'interval_cv', 'interval_iqr_median_ratio']].copy()
    result.columns = ['wallet_id', 'timing_regularity_score', 'interval_cv', 'interval_iqr_median_ratio']
    
    return result

def feature_behavior_tx_sequence_autocorrelation(open_df, tx_df, close_df):
    """
    交易序列自相关特征：衡量交易金额和间隔的时间序列自相关性
    基于领域知识：
    - 洗钱交易的金额序列呈现"先小后大"（测试→大额）模式
    - 正常交易的金额序列更随机
    - "验卡行为：正式跑分前有小额测试交易"
    
    通过金额和间隔的lag-1自相关系数来识别序列模式异常
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['amount_abs'] = tx['amount'].abs()
    tx = tx.sort_values(['src', 'timestamp']).reset_index(drop=True)
    
    def calc_autocorr(group):
        amounts = group['amount_abs'].values
        intervals = group['interval'].values
        n = len(amounts)
        
        if n < 3:
            return 0, 0, 0, 0
        
        # 金额自相关
        amt_mean = np.mean(amounts)
        amt_std = np.std(amounts)
        if amt_std > 0:
            amt_norm = (amounts - amt_mean) / amt_std
            amt_autocorr = np.mean(amt_norm[:-1] * amt_norm[1:])
        else:
            amt_autocorr = 0
        
        # 间隔自相关
        int_mean = np.mean(intervals)
        int_std = np.std(intervals)
        if int_std > 0:
            int_norm = (intervals - int_mean) / int_std
            int_autocorr = np.mean(int_norm[:-1] * int_norm[1:])
        else:
            int_autocorr = 0
        
        # 金额变化趋势（最后3笔 vs 前3笔）
        if n >= 6:
            first_3 = np.mean(amounts[:3])
            last_3 = np.mean(amounts[-3:])
            trend = (last_3 - first_3) / (first_3 + 1)
        else:
            trend = 0
        
        # 金额的"测试→大额"模式得分
        # 如果前几笔金额小后几笔金额大，则有测试模式
        if n >= 4:
            first_half = np.mean(amounts[:n//2])
            second_half = np.mean(amounts[n//2:])
            test_pattern = (second_half - first_half) / (first_half + 1)
        else:
            test_pattern = 0
        
        return amt_autocorr, int_autocorr, trend, test_pattern

        lambda g: pd.Series(calc_autocorr(g),

        features['test_pattern'].clip(0, 10) / 10 * 0.5  # 测试模式得分
    )
    
    result = features[['src', 'sequence_autocorr_score', 'amt_autocorr', 'test_pattern']].copy()
    result.columns = ['wallet_id', 'sequence_autocorr_score', 'amt_autocorr', 'test_pattern']
    
    return result
