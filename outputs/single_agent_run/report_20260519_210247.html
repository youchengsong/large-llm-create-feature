amount_concentration_pattern — 该特征检测钱包中交易金额是否集中在特定值附近。洗钱交易常因以下原因导致金额集中： 
def feature_amount_concentration_pattern(open_df, tx_df, close_df):
    """
    金额集中度模式 — 检测钱包中金额是否集中在特定值附近
    洗钱交易常因限额控制（如日累计额度）或佣金抽成导致金额集中在特定值附近，
    而正常交易的金额分布更分散。
    """
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    df = tx_df[['src', 'amount']].copy()
    
    # 对金额取整到不同粒度
    df['amount_round_100'] = (df['amount'] / 100).round() * 100
    df['amount_round_500'] = (df['amount'] / 500).round() * 500
    df['amount_round_1000'] = (df['amount'] / 1000).round() * 1000
    
    def compute_concentration(group):
        """计算金额集中度指标"""
        n = len(group)
        if n < 3:
            return 0.0
        
        # 1. 金额在百元整数倍上的集中度
        top1_100 = group['amount_round_100'].value_counts().iloc[0] if len(group['amount_round_100']) > 0 else 0
        conc_100 = top1_100 / n
        
        # 2. 金额在500元整数倍上的集中度
        top1_500 = group['amount_round_500'].value_counts().iloc[0] if len(group['amount_round_500']) > 0 else 0
        conc_500 = top1_500 / n
        
        # 3. 金额在1000元整数倍上的集中度
        top1_1000 = group['amount_round_1000'].value_counts().iloc[0] if len(group['amount_round_1000']) > 0 else 0
        conc_1000 = top1_1000 / n
        
        # 4. 金额的变异系数（越小说明金额越集中）
        cv = group['amount'].std() / (group['amount'].mean() + 0.01)
        cv_score = 1 / (1 + cv)  # CV越小，分数越高
        
        # 综合集中度分数
        score = 0.3 * conc_100 + 0.3 * conc_500 + 0.2 * conc_1000 + 0.2 * cv_score
        return min(score, 1.0)
    
    # 按src分组计算
    grp = df.groupby('src')
    result = grp.apply(compute_concentration).reset_index()
    result.columns = ['wallet_id', 'amount_concentration_pattern']
    
    return result

推理: 从分布差异发现：异常交易中regular_large_low_freq占比69.7%（正常59.9%），且异常交易金额中位数17410.54（正常10660.80）。领域知识提到"四类钱包洗钱：利用限额特征进行资金清洗，存在明显的日累计额度控制特征"以及"商户洗钱：伴随一定佣金抽成特征"。该特征通过分析钱包内金额的集中度模式——是否大量交易集中在某个特定金额附近（如日限额、佣金比例后的金额），来识别洗钱行为。

region_night_risk_score — 该特征结合钱包所在region的风险水平和夜间交易倾向。高风险region中异常 
def feature_region_night_risk_score(open_df, tx_df, close_df):
    """
    地域夜间风险分数 — 结合钱包所属region的风险水平和夜间交易倾向
    高风险region中异常交易夜间占比11.78%（正常仅3.78%），
    夜间交易+高风险region的组合显著提升洗钱概率。
    """
    import pandas as pd
    import numpy as np
    
    # 1. 计算每个region的风险水平（基于异常交易占比）
    src_region = open_df[['wallet_id', 'region_code']].copy()
    src_region.columns = ['src', 'src_region']
    tx_with_region = tx_df.merge(src_region, on='src', how='left')
    
    # 计算每个region的异常交易率
    region_risk_rate = tx_with_region.groupby('src_region')['is_risk'].mean().reset_index()
    region_risk_rate.columns = ['region_code', 'region_risk_rate']
    
    # 2. 计算每个钱包的夜间交易占比
    df = tx_df[['src', 'hour']].copy()
    df['is_night'] = ((df['hour'] >= 22) | (df['hour'] <= 5)).astype(int)
    df['is_dawn'] = ((df['hour'] >= 0) & (df['hour'] <= 5)).astype(int)
    df['is_late_night'] = ((df['hour'] >= 23) | (df['hour'] <= 3)).astype(int)
    
    wallet_night = df.groupby('src').agg(
        night_ratio=('is_night', 'mean'),
        dawn_ratio=('is_dawn', 'mean'),
        late_night_ratio=('is_late_night', 'mean'),
    ).reset_index()
    wallet_night.columns = ['wallet_id', 'night_ratio', 'dawn_ratio', 'late_night_ratio']
    
    # 3. 关联钱包的region
    wallet_region = open_df[['wallet_id', 'region_code']].copy()
    wallet_region.columns = ['wallet_id', 'region_code']
    
    result = wallet_night.merge(wallet_region, on='wallet_id', how='left')
    result = result.merge(region_risk_rate, on='region_code', how='left')
    
    # 4. 综合分数：夜间交易占比 * (1 + region风险率)
    result['region_night_risk_score'] = (
        result['night_ratio'] * (1 + result['region_risk_rate'] * 5) +
        result['dawn_ratio'] * 0.5 +
        result['late_night_ratio'] * 0.3
    )
    
    result = result[['wallet_id', 'region_night_risk_score']]
    
    return result

result = feature_region_night_risk_score(open_df, tx_df, close_df)

fan_in_without_single_pattern — 该特征检测钱包是否使用fan_in（资金汇聚）但从不使用single（单笔交易） 
def feature_fan_in_without_single_pattern(open_df, tx_df, close_df):
    """
    有fan_in无single模式 — 检测钱包是否使用fan_in但从不使用single
    异常钱包"有fan_in但无single"的比例18.5%（正常0.6%），差异30倍！
    洗钱钱包使用fan_in（资金汇聚）但从不使用single（单笔交易）。
    """
    import pandas as pd
    import numpy as np
    
    df = tx_df[['src', 'transaction_mode']].copy()
    
    # 每个钱包使用的模式集合
    mode_sets = df.groupby('src')['transaction_mode'].apply(set).reset_index()
    mode_sets.columns = ['wallet_id', 'mode_set']
    
    # 是否有fan_in
    mode_sets['has_fan_in'] = mode_sets['mode_set'].apply(lambda s: 'fan_in' in s).astype(float)
    
    # 是否有single
    mode_sets['has_single'] = mode_sets['mode_set'].apply(lambda s: 'single' in s).astype(float)
    
    # 是否有fan_out
    mode_sets['has_fan_out'] = mode_sets['mode_set'].apply(lambda s: 'fan_out' in s).astype(float)
    
    # 是否有forward
    mode_sets['has_forward'] = mode_sets['mode_set'].apply(lambda s: 'forward' in s).astype(float)
    
    # 是否有many_to_one
    mode_sets['has_many_to_one'] = mode_sets['mode_set'].apply(lambda s: 'many_to_one' in s).astype(float)
    
    # 综合模式风险分数
    # 高：有fan_in + 无single + 有fan_out
    # 低：有single
    score = (
        0.4 * mode_sets['has_fan_in'] +
        0.3 * (1 - mode_sets['has_single']) +
        0.2 * mode_sets['has_fan_out'] +
        0.1 * mode_sets['has_forward']
    )
    
    result = pd.DataFrame({
        'wallet_id': mode_sets['wallet_id'],
        'fan_in_without_single_pattern': score.values
    })
    
    return result

推理: 从数据发现：异常钱包"有fan_in但无single"的比例18.5%（正常0.6%），差异30倍！领域知识提到"洗钱交易模式包含扇入、扇出等复杂资金流转模式"、"个人钱包收到大额转入后立即转出"。正常钱包通常有single模式（单笔交易），而洗钱钱包使用fan_in（资金汇聚）但从不使用single模式。该特征检测这种模式组合——有fan_in但无single是强洗钱信号。
