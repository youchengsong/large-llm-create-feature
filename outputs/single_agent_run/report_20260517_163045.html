night_operation_intensity — 计算源钱包的夜间交易比例（0-5点）、凌晨操作比例（22点-4点）、夜间金额占比 
def feature_night_operation_intensity(open_df, tx_df, close_df):
    """计算夜间交易强度——洗钱常集中在凌晨/深夜时段操作"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 按源钱包分组
    src_grp = tx.groupby('src')
    
    def calc_night_stats(grp):
        hours = grp['hour'].values
        total = len(hours)
        if total == 0:
            return pd.Series({
                'night_tx_ratio': 0, 'dawn_tx_ratio': 0,
                'night_tx_count': 0, 'night_amount_ratio': 0,
                'hour_entropy': 0, 'peak_hour': 0,
                'night_avg_amount': 0, 'day_avg_amount_ratio': 0
            })
        
        # 深夜交易比例（0-5点）
        night_count = np.sum((hours >= 0) & (hours <= 5))
        night_ratio = night_count / total if total > 0 else 0
        
        # 凌晨-清晨（22点-次日4点）
        dawn_count = np.sum((hours >= 22) | (hours <= 4))
        dawn_ratio = dawn_count / total if total > 0 else 0
        
        # 夜间交易金额占比
        night_mask = (hours >= 0) & (hours <= 5)
        night_amounts = grp.loc[night_mask, 'amount'].sum() if night_mask.sum() > 0 else 0
        total_amount = grp['amount'].sum()
        night_amount_ratio = night_amounts / total_amount if total_amount > 0 else 0
        
        # 交易小时熵值
        hour_counts = np.zeros(24)
        for h in hours:
            hour_counts[int(h)] += 1
        probs = hour_counts / total
        entropy = -np.sum(probs[probs > 0] * np.log2(probs[probs > 0]))
        max_entropy = np.log2(24)
        hour_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        # 峰值小时
        peak_hour = np.argmax(hour_counts)
        
        # 夜间平均金额 vs 白天平均金额
        day_mask = ~night_mask
        night_avg = grp.loc[night_mask, 'amount'].mean() if night_mask.sum() > 0 else 0
        day_avg = grp.loc[day_mask, 'amount'].mean() if day_mask.sum() > 0 else 1
        day_night_ratio = night_avg / day_avg if day_avg > 0 else 0
        
        return pd.Series({
            'night_tx_ratio': night_ratio,
            'dawn_tx_ratio': dawn_ratio,
            'night_tx_count': night_count,
            'night_amount_ratio': night_amount_ratio,
            'hour_entropy': hour_entropy,
            'peak_hour': peak_hour,
            'night_avg_amount': night_avg,
            'day_avg_amount_ratio': day_night_ratio
        })
    
    result = src_grp.apply(calc_night_stats).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 裁判文书和专家访谈均指出"凌晨高频交易（20时-1时）"是线上跑分的标准模式，"夜班集中操作（22:00-04:00）是大额跑分的标准模式，单晚可达100万+"。凌晨(0-5时)也有显著活动（14%），对应在线洗钱模式。

def feature_transaction_mode_sequence_entropy(open_df, tx_df, close_df):
    """分析交易模式序列的多样性和转换模式——洗钱交易模式单一重复，正常用户模式更多样"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    def calc_mode_stats(grp):
        modes = grp['transaction_mode'].values
        motifs = grp['transaction_motif'].values if 'transaction_motif' in grp.columns else np.array([''] * len(modes))
        total = len(modes)
        
        if total == 0:
            return pd.Series({
                'mode_entropy': 0, 'mode_change_ratio': 0,
                'dominant_mode_ratio': 0, 'unique_mode_count': 0,
                'fan_out_sequence_ratio': 0, 'fan_in_sequence_ratio': 0
            })
        
        # 交易模式熵值（多样性）
        unique_modes, mode_counts = np.unique(modes, return_counts=True)
        probs = mode_counts / total
        entropy = -np.sum(probs * np.log2(probs))
        max_entropy = np.log2(len(unique_modes)) if len(unique_modes) > 1 else 1
        mode_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        # 主导模式占比（最常用模式的比例）
        dominant_mode_ratio = np.max(mode_counts) / total
        
        # 模式变化比例（相邻交易模式不同的比例）
        if total >= 2:
            mode_changes = np.sum(modes[1:] != modes[:-1])
            mode_change_ratio = mode_changes / (total - 1)
        else:
            mode_change_ratio = 0
        
        # 扇出模式占比（fan_out）
        fan_out_ratio = np.sum(modes == 'fan_out') / total if total > 0 else 0
        
        # 扇入模式占比（fan_in）
        fan_in_ratio = np.sum(modes == 'fan_in') / total if total > 0 else 0
        
        # 复杂模式占比（many_to_many, one_to_many等）
        complex_modes = np.sum(np.isin(modes, ['many_to_many', 'one_to_many', 'many_to_one']))
        complex_ratio = complex_modes / total if total > 0 else 0
        
        return pd.Series({
            'mode_entropy': mode_entropy,
            'mode_change_ratio': mode_change_ratio,
            'dominant_mode_ratio': dominant_mode_ratio,
            'unique_mode_count': len(unique_modes),
            'fan_out_ratio': fan_out_ratio,
            'fan_in_ratio': fan_in_ratio,
            'complex_mode_ratio': complex_ratio
        })
    
    result = tx.groupby('src').apply(calc_mode_stats).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 交易模式说明显示洗钱模式中"以扇出为主"（小额测试交易）、"以一对多交易为主"（商户洗钱）、"包含扇入、扇出、多对一、一对多及多对多等多种复杂结构"（在线洗钱）。洗钱通常有固定的模式偏好，模式熵值低。

def feature_tx_interval_velocity(open_df, tx_df, close_df):
    """计算交易间隔的统计特征和资金流转速度——洗钱的核心特征是资金快速进出"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    # 计算每个源钱包的交易间隔统计
    def calc_interval_stats(grp):
        if len(grp) < 2:
            return pd.Series({
                'mean_interval': 0, 'std_interval': 0, 'min_interval': 0,
                'median_interval': 0, 'interval_cv': 0,
                'rapid_tx_ratio': 0, 'total_duration_hours': 0,
                'tx_per_hour': 0
            })
        intervals = grp['interval'].values[1:]  # 第一个间隔为0，跳过
        if len(intervals) == 0:
            return pd.Series({
                'mean_interval': 0, 'std_interval': 0, 'min_interval': 0,
                'median_interval': 0, 'interval_cv': 0,
                'rapid_tx_ratio': 0, 'total_duration_hours': 0,
                'tx_per_hour': 0
            })
        
        mean_int = np.mean(intervals)
        std_int = np.std(intervals)
        min_int = np.min(intervals)
        median_int = np.median(intervals)
        
        # 变异系数
        cv = std_int / mean_int if mean_int > 0 else 0
        
        # 快速交易比例（间隔<60秒的交易占比）- 洗钱常快速连续操作
        rapid_ratio = np.sum(intervals < 60) / len(intervals)
        
        # 总持续时间（小时）
        total_sec = grp['timestamp'].iloc[-1] - grp['timestamp'].iloc[0]
        total_hours = total_sec.total_seconds() / 3600 if total_sec.total_seconds() > 0 else 0
        
        # 每小时交易次数
        tx_per_hour = len(grp) / total_hours if total_hours > 0 else len(grp)
        
        return pd.Series({
            'mean_interval': mean_int, 'std_interval': std_int, 'min_interval': min_int,
            'median_interval': median_int, 'interval_cv': cv,
            'rapid_tx_ratio': rapid_ratio, 'total_duration_hours': total_hours,
            'tx_per_hour': tx_per_hour
        })
    
    result = tx.groupby('src').apply(calc_interval_stats).reset_index()
    result.columns = ['wallet_id', 'mean_interval', 'std_interval', 'min_interval',
                      'median_interval', 'interval_cv', 'rapid_tx_ratio',
                      'total_duration_hours', 'tx_per_hour']
    
    # 资金流转速度指标：总金额/总时间（元/小时）
    src_amount_sum = tx.groupby('src')['amount'].sum().reset_index()
    src_amount_sum.columns = ['wallet_id', 'total_amount']
    result = result.merge(src_amount_sum, on='wallet_id', how='left')
    
    result['amount_flow_rate'] = result['total_amount'] / result['total_duration_hours'].clip(lower=0.01)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result[['wallet_id', 'mean_interval', 'min_interval', 'median_interval',
                   'interval_cv', 'rapid_tx_ratio', 'tx_per_hour', 'total_duration_hours',
                   'amount_flow_rate']]

推理: 裁判文书明确指出"当日即转即取"（58/309案件），资金入账后极短时间内全部转出，最快75秒完成出账。快速交易比例高+间隔变异系数低（规律性强）是洗钱的典型特征。

def feature_amount_rolling_volatility(open_df, tx_df, close_df):
    """计算交易金额的滚动波动率和金额突变检测——洗钱交易金额常突然大幅变化"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    def calc_volatility_stats(grp):
        amounts = grp['amount'].values
        total = len(amounts)
        if total < 3:
            return pd.Series({
                'amount_volatility': 0, 'max_jump_ratio': 0,
                'mean_jump_ratio': 0, 'jump_std': 0,
                'high_jump_ratio': 0, 'amount_skew': 0
            })
        
        # 计算相邻交易金额的变化率
        jumps = np.abs(amounts[1:] - amounts[:-1])
        jump_ratios = jumps / (np.maximum(amounts[:-1], 0.01))
        
        # 最大跳变比例
        max_jump = np.max(jump_ratios) if len(jump_ratios) > 0 else 0
        
        # 平均跳变比例
        mean_jump = np.mean(jump_ratios) if len(jump_ratios) > 0 else 0
        
        # 跳变标准差
        jump_std = np.std(jump_ratios) if len(jump_ratios) > 0 else 0
        
        # 高跳变比例（跳变>5倍）
        high_jump_ratio = np.sum(jump_ratios > 5) / len(jump_ratios) if len(jump_ratios) > 0 else 0
        
        # 金额波动率（变异系数）
        amount_vol = np.std(amounts) / np.mean(amounts) if np.mean(amounts) > 0 else 0
        
        # 金额偏度（洗钱金额分布可能偏大或偏小）
        if len(amounts) >= 3:
            from scipy import stats
            amount_skew = stats.skew(amounts)
        else:
            amount_skew = 0
        
        return pd.Series({
            'amount_volatility': amount_vol,
            'max_jump_ratio': max_jump,
            'mean_jump_ratio': mean_jump,
            'jump_std': jump_std,
            'high_jump_ratio': high_jump_ratio,
            'amount_skew': amount_skew
        })
    
    result = tx.groupby('src').apply(calc_volatility_stats).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    # 金额波动率与跳变的组合指标
    result['volatility_risk_score'] = result['amount_volatility'] * result['high_jump_ratio']
    
    result.fillna(0, inplace=True)

hourly_amount_profile_deviation — 检测每小时金额分布偏差：深夜/白天金额比、小时分布偏差（与全局基准的余弦距离）、 
def feature_hourly_amount_profile_deviation(open_df, tx_df, close_df):
    """检测每小时金额分布偏差——洗钱深夜平均金额反常地高"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 全局各小时的平均金额
    global_hourly_avg = tx.groupby('hour')['amount'].mean().to_dict()
    
    # 按源钱包计算
    def calc_hourly_profile(grp):
        hours = grp['hour'].values
        amounts = grp['amount'].values
        total = len(hours)
        
        if total < 3:
            return pd.Series([0, 0, 0, 0],
                           index=['night_amount_anomaly', 'hourly_profile_deviation',
                                  'peak_hour_amount_ratio', 'offpeak_amount_ratio'])
        
        # 各小时金额
        wallet_hourly = {}
        for h, a in zip(hours, amounts):
            if h not in wallet_hourly:
                wallet_hourly[h] = []
            wallet_hourly[h].append(a)
        
        # 深夜/白天金额比
        night_amounts = []
        day_amounts = []
        for h in range(24):
            if h in wallet_hourly:
                avg = np.mean(wallet_hourly[h])
                if 0 <= h <= 5:
                    night_amounts.append(avg)
                elif 8 <= h <= 20:
                    day_amounts.append(avg)
        
        night_avg = np.mean(night_amounts) if night_amounts else 0
        day_avg = np.mean(day_amounts) if day_amounts else 1
        night_amount_anomaly = night_avg / day_avg if day_avg > 0 else 0
        
        # 与全局分布偏差
        wallet_dist = np.zeros(24)
        for h in range(24):
            if h in wallet_hourly:
                wallet_dist[h] = len(wallet_hourly[h])
        wallet_probs = wallet_dist / total if total > 0 else wallet_dist
        
        global_dist = np.array([global_hourly_avg.get(h, 0) for h in range(24)])
        global_probs = global_dist / global_dist.sum() if global_dist.sum() > 0 else np.ones(24)/24
        
        if np.linalg.norm(wallet_probs) > 0 and np.linalg.norm(global_probs) > 0:
            cos_sim = np.dot(wallet_probs, global_probs) / (np.linalg.norm(wallet_probs) * np.linalg.norm(global_probs))
            deviation = 1 - cos_sim
        else:
            deviation = 0
        
        # 高峰/非高峰时段金额占比
        peak_hours_set = {10, 11, 14, 15, 16, 20, 21}
        offpeak_hours_set = {0, 1, 2, 3, 4, 5, 22, 23}
        
        peak_amount = sum(amounts[i] for i in range(total) if hours[i] in peak_hours_set)
        offpeak_amount = sum(amounts[i] for i in range(total) if hours[i] in offpeak_hours_set)
        total_amount = sum(amounts)
        
        peak_ratio = peak_amount / total_amount if total_amount > 0 else 0
        offpeak_ratio = offpeak_amount / total_amount if total_amount > 0 else 0
        
        return pd.Series([night_amount_anomaly, deviation, peak_ratio, offpeak_ratio],
                        index=['night_amount_anomaly', 'hourly_profile_deviation',
                               'peak_hour_amount_ratio', 'offpeak_amount_ratio'])
    
    result = tx.groupby('src').apply(calc_hourly_profile).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

def feature_tx_amount_momentum(open_df, tx_df, close_df):

    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    def calc_momentum(grp):

        n = len(amounts)
        
        if n < 4:
            return pd.Series({

                'amount_acceleration': 0,
                'upward_trend_strength': 0,
                'downward_trend_strength': 0,
                'amount_range_ratio': 0
            })
        
        # 动量：最近3笔平均 - 最早3笔平均
        if n >= 6:
            recent_avg = np.mean(amounts[-3:])
            early_avg = np.mean(amounts[:3])
            momentum = (recent_avg - early_avg) / (early_avg + 0.01)
        else:
            half = n // 2
            recent_avg = np.mean(amounts[half:])
            early_avg = np.mean(amounts[:half])
            momentum = (recent_avg - early_avg) / (early_avg + 0.01)
        
        # 加速度：动量的变化率（二阶差分）
        if n >= 5:
            diffs = np.diff(amounts)
            if len(diffs) >= 3:

            else:
                accel = 0
        else:
            accel = 0

        from scipy import stats
        trend, _ = stats.pearsonr(range(n), amounts)
        upward_strength = max(0, trend) if not np.isnan(trend) else 0
        downward_strength = max(0, -trend) if not np.isnan(trend) else 0
        
        # 金额范围比（最大-最小）/平均
        amount_range = np.max(amounts) - np.min(amounts)

        return pd.Series({
            'amount_momentum': momentum,
            'amount_acceleration': accel,

amount_interval_correlation — 检测金额与间隔的相关性：Pearson相关系数、Spearman秩相关、互信息差 
def feature_amount_interval_correlation(open_df, tx_df, close_df):
    """检测金额与间隔的相关性——洗钱金额与间隔存在固定关系"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    def calc_correlation(grp):
        amounts = grp['amount'].values
        intervals = grp['interval'].values
        n = len(amounts)
        
        if n < 5:
            return pd.Series({
                'amount_interval_corr': 0,
                'amount_interval_spearman': 0,
                'amount_interval_mutual_info': 0,
                'amount_given_interval_std': 0
            })
        
        # 有效间隔（跳过第一个0）
        valid_intervals = intervals[1:]
        valid_amounts = amounts[1:] if len(amounts) > 1 else amounts
        
        m = min(len(valid_amounts), len(valid_intervals))
        if m < 4:
            return pd.Series({
                'amount_interval_corr': 0,
                'amount_interval_spearman': 0,
                'amount_interval_mutual_info': 0,
                'amount_given_interval_std': 0
            })
        
        # Pearson相关系数
        from scipy import stats
        corr, _ = stats.pearsonr(valid_amounts[:m], valid_intervals[:m])
        corr = corr if not np.isnan(corr) else 0
        
        # Spearman秩相关
        spearman, _ = stats.spearmanr(valid_amounts[:m], valid_intervals[:m])
        spearman = spearman if not np.isnan(spearman) else 0
        
        # 将间隔离散化后，金额的条件标准差
        # 短间隔（<60秒）的金额 vs 长间隔的金额
        short_mask = valid_intervals[:m] < 60
        if np.sum(short_mask) > 2 and np.sum(~short_mask) > 2:
            short_amounts = valid_amounts[:m][short_mask]
            long_amounts = valid_amounts[:m][~short_mask]
            # 条件标准差差异
            short_std = np.std(short_amounts)
            long_std = np.std(long_amounts)
            cond_std_diff = abs(short_std - long_std) / (np.mean([short_std, long_std]) + 0.01)
        else:
            cond_std_diff = 0
        
        return pd.Series({
            'amount_interval_corr': corr,
            'amount_interval_spearman': spearman,
            'amount_interval_mutual_info': abs(corr - spearman),
            'amount_given_interval_std': cond_std_diff
        })
    
    result = tx.groupby('src').apply(calc_correlation).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 洗钱操作中，操作员快速连续转出大额资金（金额大+间隔短），而正常用户的大额交易往往间隔较长（需要准备时间）。金额与间隔的负相关是洗钱的重要序列特征。

hourly_amount_profile_deviation — 检测每小时金额分布偏差：深夜/白天金额比、小时分布偏差（与全局基准的余弦距离）、 
def feature_hourly_amount_profile_deviation(open_df, tx_df, close_df):
    """检测每小时金额分布偏差——洗钱深夜平均金额反常地高"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 全局各小时的平均金额
    global_hourly_avg = tx.groupby('hour')['amount'].mean().to_dict()
    
    # 按源钱包计算
    def calc_hourly_profile(grp):
        hours = grp['hour'].values
        amounts = grp['amount'].values
        total = len(hours)
        
        if total < 3:
            return pd.Series([0, 0, 0, 0],
                           index=['night_amount_anomaly', 'hourly_profile_deviation',
                                  'peak_hour_amount_ratio', 'offpeak_amount_ratio'])
        
        # 各小时金额
        wallet_hourly = {}
        for h, a in zip(hours, amounts):
            if h not in wallet_hourly:
                wallet_hourly[h] = []
            wallet_hourly[h].append(a)
        
        # 深夜/白天金额比
        night_amounts = []
        day_amounts = []
        for h in range(24):
            if h in wallet_hourly:
                avg = np.mean(wallet_hourly[h])
                if 0 <= h <= 5:
                    night_amounts.append(avg)
                elif 8 <= h <= 20:
                    day_amounts.append(avg)
        
        night_avg = np.mean(night_amounts) if night_amounts else 0
        day_avg = np.mean(day_amounts) if day_amounts else 1
        night_amount_anomaly = night_avg / day_avg if day_avg > 0 else 0
        
        # 与全局分布偏差
        wallet_dist = np.zeros(24)
        for h in range(24):
            if h in wallet_hourly:
                wallet_dist[h] = len(wallet_hourly[h])
        wallet_probs = wallet_dist / total if total > 0 else wallet_dist
        
        global_dist = np.array([global_hourly_avg.get(h, 0) for h in range(24)])
        global_probs = global_dist / global_dist.sum() if global_dist.sum() > 0 else np.ones(24)/24
        
        if np.linalg.norm(wallet_probs) > 0 and np.linalg.norm(global_probs) > 0:
            cos_sim = np.dot(wallet_probs, global_probs) / (np.linalg.norm(wallet_probs) * np.linalg.norm(global_probs))
            deviation = 1 - cos_sim
        else:
            deviation = 0
        
        # 高峰/非高峰时段金额占比
        peak_hours_set = {10, 11, 14, 15, 16, 20, 21}
        offpeak_hours_set = {0, 1, 2, 3, 4, 5, 22, 23}
        
        peak_amount = sum(amounts[i] for i in range(total) if hours[i] in peak_hours_set)
        offpeak_amount = sum(amounts[i] for i in range(total) if hours[i] in offpeak_hours_set)
        total_amount = sum(amounts)
        
        peak_ratio = peak_amount / total_amount if total_amount > 0 else 0
        offpeak_ratio = offpeak_amount / total_amount if total_amount > 0 else 0
        
        return pd.Series([night_amount_anomaly, deviation, peak_ratio, offpeak_ratio],
                        index=['night_amount_anomaly', 'hourly_profile_deviation',
                               'peak_hour_amount_ratio', 'offpeak_amount_ratio'])
    
    result = tx.groupby('src').apply(calc_hourly_profile).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 正常用户在深夜的交易通常金额很小，而洗钱在深夜的交易金额可能很大（大额跑分）。深夜金额与白天金额的比值异常高是洗钱的重要特征。与全局分布的偏差也反映了行为异常。

tx_sequence_consistency_score — 检测交易序列的一致性：金额一致性、模式一致性、间隔一致性、综合一致性。洗钱交易高 
def feature_tx_sequence_consistency_score(open_df, tx_df, close_df):
    """检测交易序列的一致性——洗钱交易模式高度一致（同质化）"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    def calc_consistency(grp):
        amounts = grp['amount'].values
        intervals = grp['interval'].values
        modes = grp['transaction_mode'].values
        n = len(amounts)
        
        if n < 4:
            return pd.Series({
                'amount_consistency': 0,
                'mode_consistency': 0,
                'interval_consistency': 0,
                'overall_consistency': 0
            })
        
        # 金额一致性：金额是否都接近（低变异系数）
        if np.mean(amounts) > 0:
            amount_cv = np.std(amounts) / np.mean(amounts)
            amount_consistency = 1 / (1 + amount_cv)
        else:
            amount_consistency = 0
        
        # 模式一致性：是否都用同一种交易模式
        unique_modes = len(np.unique(modes))
        mode_consistency = 1 - ((unique_modes - 1) / max(len(np.unique(modes)), 1))
        
        # 间隔一致性
        valid_intervals = intervals[1:]
        if len(valid_intervals) > 1 and np.mean(valid_intervals) > 0:
            interval_cv = np.std(valid_intervals) / np.mean(valid_intervals)
            interval_consistency = 1 / (1 + interval_cv)
        else:
            interval_consistency = 0
        
        # 综合一致性
        overall = (amount_consistency + mode_consistency + interval_consistency) / 3
        
        return pd.Series({
            'amount_consistency': amount_consistency,
            'mode_consistency': mode_consistency,
            'interval_consistency': interval_consistency,
            'overall_consistency': overall
        })
    
    result = tx.groupby('src').apply(calc_consistency).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 洗钱操作员按固定指令执行转账，金额、模式、间隔高度一致（同质化）。正常用户的交易行为更加多样化（不同金额、不同模式、不同间隔）。综合一致性得分高是洗钱操作机械化特征的体现。

def feature_tx_amount_sequence_complexity(open_df, tx_df, close_df):

tx_mode_switch_frequency — 检测交易模式切换频率：模式切换次数、切换率、模式持久性、转换熵值。洗钱在特定模式 
def feature_tx_mode_switch_frequency(open_df, tx_df, close_df):
    """检测交易模式切换频率——洗钱在特定模式间频繁切换"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx = tx.sort_values(['src', 'timestamp'])
    
    # 简化模式分类
    mode_map = {
        'single_transaction': 0,
        'fan_out': 1,
        'fan_in': 2,
        'one_to_many': 3,
        'many_to_one': 4,
        'many_to_many': 5
    }
    tx['mode_code'] = tx['transaction_mode'].map(mode_map).fillna(0).astype(int)
    
    def calc_switch_freq(grp):
        modes = grp['mode_code'].values
        n = len(modes)
        
        if n < 3:
            return pd.Series({
                'mode_switch_count': 0,
                'mode_switch_rate': 0,
                'mode_persistence': 0,
                'mode_transition_entropy': 0
            })
        
        # 模式切换次数
        switches = np.sum(modes[1:] != modes[:-1])
        switch_rate = switches / (n - 1) if n > 1 else 0
        
        # 模式持久性（连续相同模式的最大长度/n）
        max_run = 1
        current_run = 1
        for i in range(1, n):
            if modes[i] == modes[i-1]:
                current_run += 1
                max_run = max(max_run, current_run)
            else:
                current_run = 1
        persistence = max_run / n if n > 0 else 0
        
        # 模式转换熵值（转换矩阵的熵）
        # 统计各种转换对的出现次数
        if n >= 2:
            transitions = {}
            for i in range(n - 1):
                key = (modes[i], modes[i+1])
                transitions[key] = transitions.get(key, 0) + 1
            total_trans = sum(transitions.values())
            probs = np.array(list(transitions.values())) / total_trans
            trans_entropy = -np.sum(probs * np.log2(probs))
            max_entropy = np.log2(len(probs)) if len(probs) > 1 else 1
            transition_entropy = trans_entropy / max_entropy if max_entropy > 0 else 0
        else:
            transition_entropy = 0
        
        return pd.Series({
            'mode_switch_count': switches,
            'mode_switch_rate': switch_rate,
            'mode_persistence': persistence,
            'mode_transition_entropy': transition_entropy
        })
    
    result = tx.groupby('src').apply(calc_switch_freq).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 洗钱操作中模式切换有固定规律（如先fan_out分散再fan_in集中），转换模式固定。正常用户的模式转换更随机。模式转换熵值低+切换率高是洗钱操作的典型特征。

tx_reciprocal_network_features — 检测互惠交易网络特征：出度、入度、度比、同时是源和目标标记。洗钱网络中钱包角色单 
def feature_tx_reciprocal_network_features(open_df, tx_df, close_df):
    """检测互惠交易网络特征——洗钱网络中双向交易比例异常"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 计算每个钱包的出度和入度
    out_degree = tx.groupby('src')['dst'].nunique().reset_index()
    out_degree.columns = ['wallet_id', 'out_degree']
    
    in_degree = tx.groupby('dst')['src'].nunique().reset_index()
    in_degree.columns = ['wallet_id', 'in_degree']
    
    result = pd.merge(out_degree, in_degree, on='wallet_id', how='outer').fillna(0)
    result['degree_ratio'] = result['out_degree'] / result['in_degree'].clip(lower=1)
    
    # 快速计算互惠比例：检查钱包是否既是源又是目标
    # 如果一个钱包同时出现在src和dst列中，说明它可能参与了双向交易
    all_src = set(tx['src'].unique())
    all_dst = set(tx['dst'].unique())
    
    # 同时是源和目标的钱包
    both = all_src & all_dst
    
    result['is_both_src_dst'] = result['wallet_id'].isin(both).astype(float)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)

tx_amount_entropy_by_mode — 检测不同交易模式下的金额分布：模式内金额变异系数、模式间金额差异、主导模式占比。 
def feature_tx_amount_entropy_by_mode(open_df, tx_df, close_df):
    """检测不同交易模式下的金额熵值——洗钱各模式金额分布单一"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    def calc_entropy_by_mode(grp):
        n = len(grp)
        if n < 4:
            return pd.Series({
                'mode_amount_entropy': 0,
                'mode_amount_cv': 0,
                'mode_amount_range': 0,
                'single_mode_dominance': 0
            })
        
        # 按交易模式分组计算金额统计
        mode_groups = grp.groupby('transaction_mode')['amount']
        
        if len(mode_groups) < 2:
            return pd.Series({
                'mode_amount_entropy': 0,
                'mode_amount_cv': 0,
                'mode_amount_range': 0,
                'single_mode_dominance': 1.0
            })
        
        # 各模式下的金额变异系数
        cvs = []
        weights = []
        for mode, group in mode_groups:
            amounts = group.values
            if len(amounts) >= 2 and np.mean(amounts) > 0:
                cv = np.std(amounts) / np.mean(amounts)
                cvs.append(cv)
                weights.append(len(amounts))
        
        if len(cvs) == 0:
            return pd.Series({
                'mode_amount_entropy': 0,
                'mode_amount_cv': 0,
                'mode_amount_range': 0,
                'single_mode_dominance': 0
            })
        
        # 加权平均CV
        weights = np.array(weights)
        weighted_cv = np.average(cvs, weights=weights) if np.sum(weights) > 0 else 0
        
        # 模式间金额差异
        mode_means = [group.mean() for _, group in mode_groups]
        amount_range = max(mode_means) - min(mode_means) if len(mode_means) > 1 else 0
        
        # 主导模式占比
        sizes = mode_groups.size()
        dominant = sizes.max() / n
        
        return pd.Series({
            'mode_amount_entropy': weighted_cv,
            'mode_amount_cv': weighted_cv,
            'mode_amount_range': amount_range,
            'single_mode_dominance': dominant
        })
    
    result = tx.groupby('src').apply(calc_entropy_by_mode).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result[['wallet_id', 'mode_amount_cv', 'mode_amount_range', 'single_mode_dominance']]

推理: 洗钱操作在特定模式（如fan_out）下金额高度一致，模式内金额变异系数低。正常用户各模式下金额变化大。模式内金额一致性高是洗钱操作机械化特征的体现。

tx_amount_clustering_entropy — 检测金额聚集熵：金额聚集熵、最频繁金额占比、Top3金额占比、金额值密度。洗钱金 
def feature_tx_amount_clustering_entropy(open_df, tx_df, close_df):
    """检测金额聚集熵——洗钱金额集中在少数数值上"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    def calc_clustering(grp):
        amounts = grp['amount'].values
        n = len(amounts)
        
        if n < 3:
            return pd.Series({
                'amount_cluster_entropy': 0,
                'amount_mode_ratio': 0,
                'amount_top3_ratio': 0,
                'amount_value_density': 0
            })
        
        # 将金额四舍五入到整数
        rounded = np.round(amounts)
        
        # 金额值熵值
        unique_vals, counts = np.unique(rounded, return_counts=True)
        probs = counts / n
        entropy = -np.sum(probs * np.log2(probs))
        max_entropy = np.log2(len(unique_vals)) if len(unique_vals) > 1 else 1
        cluster_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        # 最频繁金额占比
        mode_ratio = np.max(counts) / n
        
        # Top3金额占比
        sorted_counts = np.sort(counts)[::-1]
        top3_ratio = np.sum(sorted_counts[:3]) / n if len(sorted_counts) >= 3 else np.sum(sorted_counts) / n
        
        # 金额值密度（唯一金额数/交易数）
        density = len(unique_vals) / n
        
        return pd.Series({
            'amount_cluster_entropy': cluster_entropy,
            'amount_mode_ratio': mode_ratio,
            'amount_top3_ratio': top3_ratio,
            'amount_value_density': density
        })
    
    result = tx.groupby('src').apply(calc_clustering).reset_index()
    result.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result.fillna(0, inplace=True)
    result.replace([np.inf, -np.inf], 0, inplace=True)
    
    return result

推理: 洗钱操作中金额常重复使用固定数值（如多次转出相同金额），金额聚集熵低。正常用户的金额分布更分散，聚集熵高。
