def feature_amount_split_ratio(open_df, tx_df, close_df):
    """检测拆分交易模式：大额资金拆分为多笔小额转出（平均2218元/笔）"""
    import pandas as pd
    import numpy as np
    
    # 按源钱包分组，统计交易金额分布
    src_grp = tx_df.groupby('src').agg(
        tx_count=('amount', 'count'),
        total_amount=('amount', 'sum'),
        mean_amount=('amount', 'mean'),
        std_amount=('amount', 'std'),
        max_amount=('amount', 'max'),
        min_amount=('amount', 'min')
    ).reset_index()
    src_grp.columns = ['wallet_id', 'tx_count', 'total_amount', 'mean_amount', 'std_amount', 'max_amount', 'min_amount']
    
    # 拆分交易特征1：如果总金额大但每笔金额小，且笔数多，说明有拆分嫌疑
    src_grp['amount_split_ratio'] = np.where(
        src_grp['total_amount'] > 0,
        src_grp['mean_amount'] / src_grp['total_amount'].clip(lower=1),
        0
    )
    
    # 拆分交易特征2：变异系数 - 金额波动大说明混合了大额和小额
    src_grp['amount_cv'] = np.where(
        (src_grp['mean_amount'] > 0) & (src_grp['std_amount'].notna()),
        src_grp['std_amount'] / src_grp['mean_amount'].clip(lower=0.01),
        0
    )
    
    # 拆分交易特征3：小额交易占比（金额<=100元的交易占比）
    small_tx_count = tx_df[tx_df['amount'] <= 100].groupby('src').size().reset_index()
    small_tx_count.columns = ['wallet_id', 'small_tx_count']
    src_grp = src_grp.merge(small_tx_count, on='wallet_id', how='left')
    src_grp['small_tx_count'] = src_grp['small_tx_count'].fillna(0)
    src_grp['small_tx_ratio'] = np.where(
        src_grp['tx_count'] > 0,
        src_grp['small_tx_count'] / src_grp['tx_count'],
        0
    )
    
    # 拆分交易特征4：接近限额的交易占比（金额在4000-5000元之间，接近4类钱包限额）
    near_limit_tx = tx_df[(tx_df['amount'] >= 4000) & (tx_df['amount'] <= 5000)].groupby('src').size().reset_index()
    near_limit_tx.columns = ['wallet_id', 'near_limit_count']
    src_grp = src_grp.merge(near_limit_tx, on='wallet_id', how='left')
    src_grp['near_limit_count'] = src_grp['near_limit_count'].fillna(0)
    src_grp['near_limit_ratio'] = np.where(
        src_grp['tx_count'] > 0,
        src_grp['near_limit_count'] / src_grp['tx_count'],
        0
    )
    
    result = src_grp[['wallet_id', 'amount_split_ratio', 'amount_cv', 'small_tx_ratio', 'near_limit_ratio']].copy()
    result = result.fillna(0)
    return result

推理: 裁判文书中明确提到"拆分多次转出（41/309案件）：大额资金拆分为多笔小额分散转出（平均2218元/笔）"，这是金额频次维度最核心的风险指标之一。同时四类钱包洗钱模式有"明显的日累计额度控制特征"，接近限额的交易是重要信号。

def feature_night_trading_intensity(open_df, tx_df, close_df):
    """检测夜间/凌晨交易模式：深夜高频交易是洗钱典型特征"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 定义夜间时段
    # 凌晨时段 0-5时（对应在线洗钱）
    # 深夜时段 22-24时（对应大额跑分）
    # 晚间时段 20-1时（对应线上跑分）
    
    tx['hour'] = tx['timestamp'].dt.hour
    
    # 各时段交易统计
    tx['is_dawn'] = ((tx['hour'] >= 0) & (tx['hour'] < 5)).astype(int)      # 凌晨
    tx['is_late_night'] = ((tx['hour'] >= 22) | (tx['hour'] < 1)).astype(int)  # 深夜
    tx['is_online_launder'] = ((tx['hour'] >= 20) | (tx['hour'] < 2)).astype(int)  # 线上跑分时段
    
    # 按源钱包统计夜间交易特征
    night_stats = tx.groupby('src').agg(
        total_tx=('amount', 'count'),
        dawn_tx=('is_dawn', 'sum'),
        late_night_tx=('is_late_night', 'sum'),
        online_launder_tx=('is_online_launder', 'sum'),
        dawn_amount=('amount', lambda x: x[tx.loc[x.index, 'is_dawn'] == 1].sum() if tx.loc[x.index, 'is_dawn'].sum() > 0 else 0),
        late_night_amount=('amount', lambda x: x[tx.loc[x.index, 'is_late_night'] == 1].sum() if tx.loc[x.index, 'is_late_night'].sum() > 0 else 0)
    ).reset_index()
    night_stats.columns = ['wallet_id', 'total_tx', 'dawn_tx', 'late_night_tx', 'online_launder_tx', 
                           'dawn_amount', 'late_night_amount']
    
    # 夜间交易占比
    night_stats['dawn_ratio'] = np.where(night_stats['total_tx'] > 0, night_stats['dawn_tx'] / night_stats['total_tx'], 0)
    night_stats['late_night_ratio'] = np.where(night_stats['total_tx'] > 0, night_stats['late_night_tx'] / night_stats['total_tx'], 0)
    night_stats['online_launder_ratio'] = np.where(night_stats['total_tx'] > 0, night_stats['online_launder_tx'] / night_stats['total_tx'], 0)
    
    # 夜间交易金额占比
    night_stats['dawn_amount_ratio'] = np.where(night_stats['total_tx'] > 0, 
        night_stats['dawn_amount'] / (night_stats.groupby('wallet_id')['dawn_amount'].transform('sum') + 1), 0)
    
    # 综合夜间异常分数
    night_stats['night_anomaly_score'] = (
        night_stats['dawn_ratio'] * 0.4 + 
        night_stats['late_night_ratio'] * 0.3 + 
        night_stats['online_launder_ratio'] * 0.3
    )
    
    result = night_stats[['wallet_id', 'dawn_ratio', 'late_night_ratio', 'online_launder_ratio', 'night_anomaly_score']].copy()
    result = result.fillna(0)
    return result

推理: 多个知识源都强调了夜间交易的重要性：交易模式中"小额测试交易主要集中在深夜或非活跃时段"、"在线洗钱更偏向夜间及深夜时段"；专家访谈中"凌晨高频交易(20时-1时)"；裁判文书中"夜班集中操作（22:00-04:00）"。

ip_region_jump — 检测IP地址和地域跳变频率。洗钱操作中，操作人员可能同时控制多个钱包在不同IP间 
def feature_ip_region_jump(open_df, tx_df, close_df):
    """检测IP地址和地域跳变：短期内跨多地域发起交易"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 提取IP的前两段作为地域粗略标识（IP段变化说明地域变化）
    tx['src_ip_prefix'] = tx['src_ip'].apply(lambda x: '.'.join(str(x).split('.')[:2]) if pd.notna(x) else 'unknown')
    tx['dst_ip_prefix'] = tx['dst_ip'].apply(lambda x: '.'.join(str(x).split('.')[:2]) if pd.notna(x) else 'unknown')
    
    # 按源钱包统计IP变化
    ip_diversity = tx.groupby('src').agg(
        unique_src_ips=('src_ip_prefix', 'nunique'),
        unique_dst_ips=('dst_ip_prefix', 'nunique'),
        total_tx=('amount', 'count')
    ).reset_index()
    ip_diversity.columns = ['wallet_id', 'unique_src_ips', 'unique_dst_ips', 'total_tx']
    
    # IP多样性比率（每笔交易使用的IP数）
    ip_diversity['src_ip_diversity_ratio'] = np.where(
        ip_diversity['total_tx'] > 0,
        ip_diversity['unique_src_ips'] / ip_diversity['total_tx'],
        0
    )
    ip_diversity['dst_ip_diversity_ratio'] = np.where(
        ip_diversity['total_tx'] > 0,
        ip_diversity['unique_dst_ips'] / ip_diversity['total_tx'],
        0
    )
    
    # 检测IP切换频率：按时间排序后统计IP变化的次数
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    tx_sorted['prev_src_ip'] = tx_sorted.groupby('src')['src_ip_prefix'].shift(1)
    tx_sorted['ip_changed'] = (tx_sorted['src_ip_prefix'] != tx_sorted['prev_src_ip']).astype(int)
    
    ip_change_count = tx_sorted.groupby('src')['ip_changed'].sum().reset_index()
    ip_change_count.columns = ['wallet_id', 'ip_change_count']
    
    ip_diversity = ip_diversity.merge(ip_change_count, on='wallet_id', how='left')
    ip_diversity['ip_change_rate'] = np.where(
        ip_diversity['total_tx'] > 1,
        ip_diversity['ip_change_count'] / (ip_diversity['total_tx'] - 1),
        0
    )
    
    # 检测源IP和目标IP是否在同一地域（异常转账往往跨地域）
    tx['same_ip_region'] = (tx['src_ip_prefix'] == tx['dst_ip_prefix']).astype(int)
    cross_region = tx.groupby('src').agg(
        cross_region_tx=('same_ip_region', lambda x: (1 - x).sum()),
        total_tx_count=('same_ip_region', 'count')
    ).reset_index()
    cross_region.columns = ['wallet_id', 'cross_region_tx', 'total_tx_count']
    cross_region['cross_region_ratio'] = np.where(
        cross_region['total_tx_count'] > 0,
        cross_region['cross_region_tx'] / cross_region['total_tx_count'],
        0
    )
    
    result = ip_diversity.merge(cross_region[['wallet_id', 'cross_region_ratio']], on='wallet_id', how='left')
    result = result[['wallet_id', 'src_ip_diversity_ratio', 'ip_change_rate', 'cross_region_ratio']].copy()
    result = result.fillna(0)
    return result

推理: 风险指标中明确提到"同一账户/关联账户的交易时间间隔"和"交易发起地（交易定位/IP）：短期内跨多地域发起交易"是重要风险指标。裁判文书中也提到操作人员通过Telegram接收指令，在多地操作多个钱包。

hour_concentration_entropy — 检测交易时间的集中度特征。洗钱操作通常集中在特定时段（如夜间2-8小时的操作窗口 
def feature_hour_concentration_entropy(open_df, tx_df, close_df):
    """检测交易时间集中度：洗钱交易通常集中在特定时段（如夜间操作窗口2-8小时）"""
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx['hour'] = tx['timestamp'].dt.hour
    
    # 按源钱包统计各小时交易分布
    def calc_hour_metrics(group):
        hours = group['hour'].values
        if len(hours) == 0:
            return pd.Series([0, 0, 0, 0, 0])
        
        # 1. 交易活跃的小时数（有多少个不同小时有交易）
        unique_hours = len(np.unique(hours))
        
        # 2. 最活跃小时占比（集中度）
        hour_counter = Counter(hours)
        most_common_hour_count = hour_counter.most_common(1)[0][1]
        peak_hour_ratio = most_common_hour_count / len(hours)
        
        # 3. 前3活跃小时占比
        top3_count = sum([c for _, c in hour_counter.most_common(3)])
        top3_hour_ratio = top3_count / len(hours) if len(hours) > 0 else 0
        
        # 4. 交易时段跨度（最早和最晚交易的小时差）
        hour_span = np.max(hours) - np.min(hours)
        # 处理跨天情况
        if hour_span < 0:
            hour_span = 24 + hour_span
        
        # 5. 小时熵值（越低说明越集中在少数时段）
        probs = np.array([c/len(hours) for c in hour_counter.values()])
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        max_entropy = np.log2(len(hour_counter)) if len(hour_counter) > 1 else 1
        normalized_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        return pd.Series([unique_hours, peak_hour_ratio, top3_hour_ratio, hour_span, normalized_entropy])
    
    result = tx.groupby('src').apply(calc_hour_metrics).reset_index()
    result.columns = ['wallet_id', 'active_hour_count', 'peak_hour_ratio', 'top3_hour_ratio', 'hour_span', 'hour_entropy']
    
    # 综合集中度异常分数：低熵+高峰值+窄时段=可疑
    result['hour_concentration_score'] = (
        (1 - result['hour_entropy']) * 0.3 + 
        result['peak_hour_ratio'] * 0.3 + 
        result['top3_hour_ratio'] * 0.2 +
        (1 - result['active_hour_count'] / 24) * 0.2
    )
    
    result = result.fillna(0)
    return result

推理: 裁判文书中提到"单次操作窗口：2-8小时完成全部资金流转"，说明洗钱交易集中在狭窄的时间窗口内。专家访谈也提到"凌晨高频交易(20时-1时)"，时间集中度是重要区分特征。

def feature_counterparty_change_frequency(open_df, tx_df, close_df):
    """检测交易对手更换频率：频繁更换陌生交易对手是洗钱典型特征"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按源钱包排序
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    # 对每个钱包，分析交易对手（dst）的变化
    def calc_counterparty_metrics(group):
        dsts = group['dst'].values
        if len(dsts) < 2:
            return pd.Series([1, 0, 0, 0, 0])  # 只有1个对手
        
        # 唯一交易对手数量
        unique_dsts = len(np.unique(dsts))
        
        # 新交易对手占比（不在前3笔中出现的对手）
        if len(dsts) <= 3:
            new_dst_ratio = 0
        else:
            known_dsts = set(dsts[:3])
            new_dsts = sum(1 for d in dsts[3:] if d not in known_dsts)
            new_dst_ratio = new_dsts / len(dsts[3:])
        
        # 交易对手切换次数（相邻交易对手不同的次数）
        switch_count = np.sum(dsts[1:] != dsts[:-1])
        switch_rate = switch_count / (len(dsts) - 1)
        
        # 交易对手熵（越低越集中在少数对手）
        from collections import Counter
        counter = Counter(dsts)
        probs = np.array([c/len(dsts) for c in counter.values()])
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        max_entropy = np.log2(len(counter)) if len(counter) > 1 else 1
        normalized_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        return pd.Series([unique_dsts, new_dst_ratio, switch_rate, normalized_entropy, len(dsts)])
    
    result = tx_sorted.groupby('src').apply(calc_counterparty_metrics).reset_index()
    result.columns = ['wallet_id', 'unique_dst_count', 'new_dst_ratio', 'dst_switch_rate', 'dst_entropy', 'total_tx']
    
    # 归一化唯一对手数
    result['dst_diversity_ratio'] = np.where(
        result['total_tx'] > 0,
        result['unique_dst_count'] / result['total_tx'],
        0
    )
    
    # 综合异常分数：高切换率+高新人比例+低熵=可疑
    result['counterparty_anomaly_score'] = (
        result['new_dst_ratio'] * 0.3 + 
        result['dst_switch_rate'] * 0.3 + 
        result['dst_diversity_ratio'] * 0.2 +
        (1 - result['dst_entropy']) * 0.2
    )
    
    result = result[['wallet_id', 'unique_dst_count', 'new_dst_ratio', 'dst_switch_rate', 'counterparty_anomaly_score']].copy()
    result = result.fillna(0)
    return result

推理: 风险指标中明确提到"频繁更换陌生交易对手"和"关联账户数量（是否存在多账户交叉转账）"。裁判文书中操作人员通过Telegram群接收上线指令转账给不同账户，交易对手不断变化。

def feature_batch_transaction_pattern(open_df, tx_df, close_df):
    """检测批量交易模式：短时间内连续多笔交易（批量操作是洗钱典型行为）"""
    import pandas as pd
    import numpy as np

def feature_hour_day_pattern_analysis(open_df, tx_df, close_df):

        is_night_peak = int((peak_hour >= 20) or (peak_hour <= 5))
        
        return pd.Series([weekend_ratio, off_hour_ratio, weekend_off, weekday_off, is_night_peak])

    result['time_anomaly_score'] = (

amount_skewness_kurtosis — 检测金额分布的高阶矩特征。洗钱交易的金额分布偏度（正偏=大量小额交易+少量大额） 
def feature_amount_skewness_kurtosis(open_df, tx_df, close_df):
    """检测金额分布的高阶矩特征：偏度和峰度"""
    import pandas as pd
    import numpy as np
    from scipy import stats
    
    tx = tx_df.copy()
    
    def calc_higher_moments(group):
        amounts = group['amount'].values
        if len(amounts) < 8:
            return pd.Series([0, 0, 0])
        
        # 1. 偏度（正偏=小额多+大额少，负偏=大额多+小额少）
        skewness = stats.skew(amounts)
        skewness = 0 if np.isnan(skewness) else skewness
        
        # 2. 峰度（高峰度=金额集中在特定值，低峰度=金额分散）
        kurt = stats.kurtosis(amounts)
        kurt = 0 if np.isnan(kurt) else kurt
        
        # 3. 金额的95分位与中位数之比（尾部厚度）
        p95 = np.percentile(amounts, 95)
        median = np.median(amounts)
        tail_ratio = p95 / median if median > 0 else 0
        
        return pd.Series([skewness, kurt, tail_ratio])
    
    result = tx.groupby('src').apply(calc_higher_moments).reset_index()
    result.columns = ['wallet_id', 'amount_skewness', 'amount_kurtosis', 'amount_tail_ratio']
    
    result = result.fillna(0)
    return result

hour_amount_profile — 检测各小时的平均金额轮廓。洗钱交易中，夜间金额可能异常高（与正常消费相反），下午 
def feature_hour_amount_profile(open_df, tx_df, close_df):
    """检测各小时的平均金额轮廓：洗钱交易的小时-金额模式"""
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx['hour'] = tx['timestamp'].dt.hour
    
    def calc_hour_amount_profile(group):
        hours = group['hour'].values
        amounts = group['amount'].values
        
        if len(amounts) < 10:
            return pd.Series([0, 0, 0, 0])
        
        # 计算每个小时的平均金额
        hour_means = np.zeros(24)
        hour_counts = np.zeros(24)
        for h, a in zip(hours, amounts):
            hour_means[int(h)] += a
            hour_counts[int(h)] += 1
        
        for h in range(24):
            hour_means[h] = hour_means[h] / hour_counts[h] if hour_counts[h] > 0 else 0
        
        # 1. 夜间(0-5时)平均金额 vs 全天平均金额
        night_mean = np.mean(hour_means[0:5]) if np.sum(hour_counts[0:5]) > 0 else 0
        overall_mean = np.mean(amounts)
        night_overall_ratio = night_mean / overall_mean if overall_mean > 0 else 0
        
        # 2. 下午(14-17时)平均金额 vs 全天平均金额
        afternoon_mean = np.mean(hour_means[14:18]) if np.sum(hour_counts[14:18]) > 0 else 0
        afternoon_overall_ratio = afternoon_mean / overall_mean if overall_mean > 0 else 0
        
        # 3. 小时平均金额的标准差（越大说明金额-时间关联越强）
        hour_mean_std = np.std(hour_means) if np.sum(hour_counts) > 0 else 0
        
        # 4. 最高金额小时 vs 最低金额小时的比值
        max_hour_mean = np.max(hour_means) if np.max(hour_means) > 0 else 1
        min_hour_mean = np.min(hour_means[hour_means > 0]) if np.any(hour_means > 0) else 1
        max_min_ratio = max_hour_mean / min_hour_mean if min_hour_mean > 0 else 0
        
        return pd.Series([night_overall_ratio, afternoon_overall_ratio, hour_mean_std, max_min_ratio])
    
    result = tx.groupby('src').apply(calc_hour_amount_profile).reset_index()
    result.columns = ['wallet_id', 'night_amount_ratio', 'afternoon_amount_ratio', 
                      'hour_amount_std', 'hour_amount_max_min_ratio']
    
    result = result.fillna(0)
    return result

推理: 裁判文书中"犯罪操作集中在下午(14-17时)最多"和"夜班集中操作（22:00-04:00）"说明不同时段金额模式不同。小时金额轮廓是时间与金额的交叉特征。

def feature_amount_interval_clustering(open_df, tx_df, close_df):
    """检测金额-间隔的联合分布模式：特定金额与特定间隔的组合"""
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按源钱包排序
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    tx_sorted['prev_time'] = tx_sorted.groupby('src')['timestamp'].shift(1)
    tx_sorted['interval_sec'] = (tx_sorted['timestamp'] - tx_sorted['prev_time']).dt.total_seconds()
    
    # 过滤
    tx_valid = tx_sorted.dropna(subset=['interval_sec']).copy()
    tx_valid = tx_valid[tx_valid['interval_sec'] >= 0]
    
    def calc_joint_distribution(group):
        amounts = group['amount'].values
        intervals = group['interval_sec'].values
        
        if len(amounts) < 5:
            return pd.Series([0, 0, 0])
        
        # 将金额和间隔离散化
        amount_bins = np.digitize(amounts, bins=[10, 50, 100, 500, 1000, 5000])
        interval_bins = np.digitize(intervals, bins=[5, 30, 60, 300, 600, 3600])
        
        # 联合分布键
        joint_keys = [f"{a}_{i}" for a, i in zip(amount_bins, interval_bins)]
        counter = Counter(joint_keys)

        top_combo_ratio = max(counter.values()) / len(joint_keys)

        probs = np.array([c/len(joint_keys) for c in counter.values()])
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        max_entropy = np.log2(len(counter)) if len(counter) > 1 else 1
        norm_entropy = entropy / max_entropy if max_entropy > 0 else 0
        
        # 3. 唯一组合数
        unique_combos = len(counter)
        combo_density = unique_combos / len(joint_keys)
        
        return pd.Series([top_combo_ratio, norm_entropy, combo_density])
    
    if len(tx_valid) > 0:
        result = tx_valid.groupby('src').apply(calc_joint_distribution).reset_index()
        result.columns = ['wallet_id', 'amount_interval_top_ratio', 'amount_interval_entropy', 'amount_interval_density']
    else:
        result = pd.DataFrame({'wallet_id': tx['src'].unique()})
        result['amount_interval_top_ratio'] = 0
        result['amount_interval_entropy'] = 0
        result['amount_interval_density'] = 0
    
    result = result.fillna(0)
    return result
