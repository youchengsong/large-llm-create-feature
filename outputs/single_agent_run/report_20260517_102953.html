def feature_rapid_outflow_ratio_v2(open_df, tx_df, close_df):
    \"\"\"即转即取比率(优化版) - 资金入账后短时间内转出的比例\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 对每个钱包，按时间排序所有交易
    # 标记每笔交易是流入(作为dst)还是流出(作为src)
    
    # 流入交易
    inflow = tx[['dst', 'timestamp', 'amount']].copy()
    inflow.columns = ['wallet_id', 'timestamp', 'amount']
    inflow['flow_type'] = 'in'
    
    # 流出交易
    outflow = tx[['src', 'timestamp', 'amount']].copy()
    outflow.columns = ['wallet_id', 'timestamp', 'amount']
    outflow['flow_type'] = 'out'
    
    # 合并并按钱包和时间排序
    all_flows = pd.concat([inflow, outflow], ignore_index=True)
    all_flows = all_flows.sort_values(['wallet_id', 'timestamp']).reset_index(drop=True)
    
    # 对每个钱包，计算每笔流入后下一次流出的时间差
    all_flows['next_timestamp'] = all_flows.groupby('wallet_id')['timestamp'].shift(-1)
    all_flows['next_type'] = all_flows.groupby('wallet_id')['flow_type'].shift(-1)
    
    # 只保留流入交易，且下一笔是流出
    inflow_with_next = all_flows[
        (all_flows['flow_type'] == 'in') & 
        (all_flows['next_type'] == 'out')
    ].copy()
    
    inflow_with_next['diff_minutes'] = (
        inflow_with_next['next_timestamp'] - inflow_with_next['timestamp']
    ).dt.total_seconds() / 60.0
    
    # 统计每个钱包
    wallet_stats = inflow_with_next.groupby('wallet_id').agg(
        rapid_count=('diff_minutes', lambda x: (x <= 60).sum()),
        total_inflow=('diff_minutes', 'count')
    ).reset_index()
    
    wallet_stats['rapid_outflow_ratio'] = (
        wallet_stats['rapid_count'] / wallet_stats['total_inflow'].clip(lower=1)
    )
    
    # 补充只有流入或只有流出的钱包
    all_wallets = pd.DataFrame({'wallet_id': pd.concat([tx['src'], tx['dst']]).unique()})
    result = all_wallets.merge(wallet_stats[['wallet_id', 'rapid_outflow_ratio']], on='wallet_id', how='left')
    result['rapid_outflow_ratio'] = result['rapid_outflow_ratio'].fillna(0)
    
    return result[['wallet_id', 'rapid_outflow_ratio']]

result = feature_rapid_outflow_ratio_v2(open_df, tx_df, close_df)

def feature_fan_out_in_ratio(open_df, tx_df, close_df):
    \"\"\"扇出扇入比 - 钱包作为源(转出)的目标数与作为目标(收款)的源数的比值\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 每个钱包作为源时，转给多少个不同的目标
    src_unique = tx.groupby('src')['dst'].nunique().reset_index()
    src_unique.columns = ['wallet_id', 'fan_out_count']
    
    # 每个钱包作为目标时，从多少个不同的源收款
    dst_unique = tx.groupby('dst')['src'].nunique().reset_index()
    dst_unique.columns = ['wallet_id', 'fan_in_count']
    
    # 合并
    result = src_unique.merge(dst_unique, on='wallet_id', how='outer').fillna(0)
    
    # 扇出/扇入比
    result['fan_out_in_ratio'] = result['fan_out_count'] / (result['fan_in_count'] + 1)
    
    # 对数变换压缩偏态
    result['fan_out_in_ratio_log'] = np.log1p(result['fan_out_in_ratio'])
    
    return result[['wallet_id', 'fan_out_in_ratio', 'fan_out_in_ratio_log', 'fan_out_count', 'fan_in_count']]

def feature_cross_bank_tx_ratio(open_df, tx_df, close_df):
    \"\"\"跨行交易比率 - 源和目标银行不同的交易占比\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 检查源和目标银行是否不同（跨行交易）
    tx['is_cross_bank'] = (
        (tx['src_bank_account_number'].notna()) & 
        (tx['dst_bank_account_number'].notna()) & 
        (tx['src_bank_account_number'] != tx['dst_bank_account_number'])
    ).astype(int)
    
    # 按源钱包统计跨行交易
    src_cross = tx.groupby('src').agg(
        src_total=('tx_id', 'count'),
        src_cross_bank=('is_cross_bank', 'sum')
    ).reset_index()
    src_cross.columns = ['wallet_id', 'src_total', 'src_cross_bank']
    src_cross['src_cross_ratio'] = src_cross['src_cross_bank'] / src_cross['src_total'].clip(lower=1)
    
    # 按目标钱包统计
    dst_cross = tx.groupby('dst').agg(
        dst_total=('tx_id', 'count'),
        dst_cross_bank=('is_cross_bank', 'sum')
    ).reset_index()
    dst_cross.columns = ['wallet_id', 'dst_total', 'dst_cross_bank']
    dst_cross['dst_cross_ratio'] = dst_cross['dst_cross_bank'] / dst_cross['dst_total'].clip(lower=1)
    
    result = src_cross[['wallet_id', 'src_cross_ratio']].merge(
        dst_cross[['wallet_id', 'dst_cross_ratio']], on='wallet_id', how='outer'
    ).fillna(0)
    
    result['cross_bank_ratio'] = result[['src_cross_ratio', 'dst_cross_ratio']].max(axis=1)
    
    return result[['wallet_id', 'cross_bank_ratio', 'src_cross_ratio']]

def feature_net_flow_ratio(open_df, tx_df, close_df):
    \"\"\"资金净流入流出比 - 检测过路账户（资金快进快出，净额接近0）\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 每个钱包作为源的总转出金额
    src_amount = tx.groupby('src')['amount'].sum().reset_index()
    src_amount.columns = ['wallet_id', 'total_outflow']
    
    # 每个钱包作为目标的总转入金额
    dst_amount = tx.groupby('dst')['amount'].sum().reset_index()
    dst_amount.columns = ['wallet_id', 'total_inflow']
    
    # 合并
    result = src_amount.merge(dst_amount, on='wallet_id', how='outer').fillna(0)
    
    # 净额
    result['net_flow'] = result['total_inflow'] - result['total_outflow']
    
    # 总流水
    result['total_volume'] = result['total_inflow'] + result['total_outflow']
    
    # 净流入/总流水比（绝对值）：过路账户接近0，资金归集账户为正，分发账户为负
    result['net_flow_ratio'] = np.abs(result['net_flow']) / result['total_volume'].clip(lower=1)
    
    # 对数变换
    result['total_volume_log'] = np.log1p(result['total_volume'])
    
    return result[['wallet_id', 'net_flow_ratio', 'net_flow', 'total_volume', 'total_volume_log']]

def feature_amount_diff_commission_pattern(open_df, tx_df, close_df):
    \"\"\"金额差额检测 - 检测转入转出金额差额是否符合佣金模式（差额约1-5%）\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 对每个钱包，按时间排序所有交易
    inflow = tx[['dst', 'timestamp', 'amount']].copy()
    inflow.columns = ['wallet_id', 'timestamp', 'in_amount']
    inflow['flow_type'] = 'in'
    
    outflow = tx[['src', 'timestamp', 'amount']].copy()
    outflow.columns = ['wallet_id', 'timestamp', 'out_amount']
    outflow['flow_type'] = 'out'
    
    # 合并并按钱包和时间排序
    all_in = inflow.sort_values(['wallet_id', 'timestamp'])
    all_out = outflow.sort_values(['wallet_id', 'timestamp'])
    
    # 对每个钱包，计算流入和流出总额
    wallet_in = all_in.groupby('wallet_id')['in_amount'].sum().reset_index()
    wallet_out = all_out.groupby('wallet_id')['out_amount'].sum().reset_index()
    
    result = wallet_in.merge(wallet_out, on='wallet_id', how='outer').fillna(0)
    
    # 流入/流出比
    result['in_out_ratio'] = result['in_amount'] / (result['out_amount'] + 1)
    result['out_in_ratio'] = result['out_amount'] / (result['in_amount'] + 1)
    
    # 差额比例（取绝对值较小的方向）
    result['diff_ratio'] = np.abs(result['in_amount'] - result['out_amount']) / (result['in_amount'] + result['out_amount'] + 1)
    
    # 佣金模式检测：流出略小于流入（扣除佣金后转出）
    # 佣金率通常在1%-5%之间，所以流出/流入在0.95-0.99之间
    result['commission_like'] = (
        (result['out_in_ratio'] >= 0.90) & 
        (result['out_in_ratio'] <= 0.995) & 
        (result['in_amount'] > 100)
    ).astype(int)
    
    # 反向佣金：流入略小于流出（收到后加价转出）
    result['reverse_commission_like'] = (
        (result['in_out_ratio'] >= 0.90) & 
        (result['in_out_ratio'] <= 0.995) & 
        (result['out_amount'] > 100)
    ).astype(int)
    
    result['commission_score'] = result['commission_like'] + result['reverse_commission_like']
    
    return result[['wallet_id', 'diff_ratio', 'in_out_ratio', 'commission_score']]

result = feature_amount_diff_commission_pattern(open_df, tx_df, close_df)

def feature_repeat_counterparty_ratio(open_df, tx_df, close_df):
    \"\"\"重复交易对手比率 - 与同一对手交易多次的占比\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 按源钱包-目标钱包对统计交易次数
    pair_counts = tx.groupby(['src', 'dst']).size().reset_index(name='pair_tx_count')
    
    # 标记重复交易（与同一对手交易>=2次）
    pair_counts['is_repeat'] = (pair_counts['pair_tx_count'] >= 2).astype(int)
    
    # 按源钱包统计
    src_repeat = pair_counts.groupby('src').agg(
        total_pairs=('dst', 'count'),
        repeat_pairs=('is_repeat', 'sum'),
        max_pair_tx=('pair_tx_count', 'max')
    ).reset_index()
    src_repeat.columns = ['wallet_id', 'total_pairs', 'repeat_pairs', 'max_pair_tx']
    src_repeat['repeat_pair_ratio'] = src_repeat['repeat_pairs'] / src_repeat['total_pairs'].clip(lower=1)
    
    # 按目标钱包统计
    dst_pair_counts = tx.groupby(['dst', 'src']).size().reset_index(name='pair_tx_count')
    dst_pair_counts['is_repeat'] = (dst_pair_counts['pair_tx_count'] >= 2).astype(int)
    
    dst_repeat = dst_pair_counts.groupby('dst').agg(
        dst_total_pairs=('src', 'count'),
        dst_repeat_pairs=('is_repeat', 'sum')
    ).reset_index()
    dst_repeat.columns = ['wallet_id', 'dst_total_pairs', 'dst_repeat_pairs']
    dst_repeat['dst_repeat_ratio'] = dst_repeat['dst_repeat_pairs'] / dst_repeat['dst_total_pairs'].clip(lower=1)
    
    result = src_repeat[['wallet_id', 'repeat_pair_ratio', 'max_pair_tx']].merge(
        dst_repeat[['wallet_id', 'dst_repeat_ratio']], on='wallet_id', how='outer'
    ).fillna(0)
    
    result['repeat_ratio_max'] = result[['repeat_pair_ratio', 'dst_repeat_ratio']].max(axis=1)
    
    return result[['wallet_id', 'repeat_ratio_max', 'max_pair_tx']]

def feature_amount_quantile_simple(open_df, tx_df, close_df):
    \"\"\"金额分位数特征(简化版) - 检测金额分布异常\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 每个钱包作为源时的金额统计
    src_stats = tx.groupby('src')['amount'].agg(
        src_q75=lambda x: x.quantile(0.75),
        src_q90=lambda x: x.quantile(0.90),
        src_max='max',
        src_min='min'
    ).reset_index()
    src_stats.columns = ['wallet_id', 'src_q75', 'src_q90', 'src_max', 'src_min']

    # 每个钱包作为目标时的金额统计
    dst_stats = tx.groupby('dst')['amount'].agg(
        dst_q75=lambda x: x.quantile(0.75),
        dst_q90=lambda x: x.quantile(0.90),
        dst_max='max',
        dst_min='min'
    ).reset_index()
    dst_stats.columns = ['wallet_id', 'dst_q75', 'dst_q90', 'dst_max', 'dst_min']
    dst_stats['dst_range'] = dst_stats['dst_max'] - dst_stats['dst_min']
    
    result = src_stats[['wallet_id', 'src_q90', 'src_range']].merge(
        dst_stats[['wallet_id', 'dst_q90', 'dst_range']], on='wallet_id', how='outer'
    ).fillna(0)
    
    result['q90_max'] = result[['src_q90', 'dst_q90']].max(axis=1)
    result['range_max'] = result[['src_range', 'dst_range']].max(axis=1)
    
    return result[['wallet_id', 'q90_max', 'range_max']]

def feature_time_distribution_entropy(open_df, tx_df, close_df):
    \"\"\"交易时间分布熵 - 检测交易时间是否集中在少数时段\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx['hour'] = tx['timestamp'].dt.hour
    
    def calc_entropy(series):
        \"\"\"计算信息熵\"\"\"
        counts = series.value_counts()
        probs = counts / len(series)
        return -sum(probs * np.log(probs + 1e-10))
    
    # 按源钱包计算时间分布熵
    src_entropy = tx.groupby('src')['hour'].agg(calc_entropy).reset_index()
    src_entropy.columns = ['wallet_id', 'src_time_entropy']
    
    # 按目标钱包计算时间分布熵
    dst_entropy = tx.groupby('dst')['hour'].agg(calc_entropy).reset_index()
    dst_entropy.columns = ['wallet_id', 'dst_time_entropy']
    
    # 计算集中度：最高频时段的交易占比
    def max_hour_ratio(series):
        return series.value_counts().max() / len(series)
    
    src_conc = tx.groupby('src')['hour'].agg(max_hour_ratio).reset_index()
    src_conc.columns = ['wallet_id', 'src_max_hour_ratio']

day_night_amount_diff — 比较白天和夜间交易金额的差异。洗钱账户夜间交易金额通常显著高于白天（夜班大额操作 
def feature_day_night_amount_diff(open_df, tx_df, close_df):
    \"\"\"白天/夜间金额差异 - 检测白天和夜间交易金额的显著差异\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx['hour'] = tx['timestamp'].dt.hour
    
    # 白天：8-18时，夜间：22-4时
    tx['period'] = 'other'
    tx.loc[tx['hour'].isin(list(range(8, 18))), 'period'] = 'day'
    tx.loc[tx['hour'].isin([22, 23, 0, 1, 2, 3, 4]), 'period'] = 'night'
    
    # 筛选有白天和夜间交易的源钱包
    day_night_tx = tx[tx['period'].isin(['day', 'night'])].copy()
    
    # 按源钱包和时段统计
    period_stats = day_night_tx.groupby(['src', 'period'])['amount'].agg(['mean', 'sum', 'count']).reset_index()
    period_stats.columns = ['wallet_id', 'period', 'avg_amount', 'total_amount', 'tx_count']
    
    # 透视
    pivot = period_stats.pivot_table(
        index='wallet_id', 
        columns='period', 
        values=['avg_amount', 'total_amount', 'tx_count'],
        fill_value=0
    )
    pivot.columns = ['_'.join(col).strip() for col in pivot.columns.values]
    pivot = pivot.reset_index()
    
    # 计算白天/夜间金额比
    result = pivot.copy()
    if 'avg_amount_day' in result.columns and 'avg_amount_night' in result.columns:
        result['day_night_amount_ratio'] = result['avg_amount_day'] / (result['avg_amount_night'] + 0.01)
        result['night_day_amount_diff'] = result['avg_amount_night'] - result['avg_amount_day']
    else:
        result['day_night_amount_ratio'] = 1.0
        result['night_day_amount_diff'] = 0.0
    
    # 补充没有昼夜区分的钱包
    all_wallets = pd.DataFrame({'wallet_id': pd.concat([tx_df['src'], tx_df['dst']]).unique()})
    result = all_wallets.merge(result, on='wallet_id', how='left').fillna(0)
    
    return result[['wallet_id', 'day_night_amount_ratio', 'night_day_amount_diff']]

wallet_centrality_simple — 简化的网络中心度特征，计算钱包的总度数（连接数）和关系强度（每对手平均交易次数） 
def feature_wallet_centrality_simple(open_df, tx_df, close_df):
    \"\"\"钱包网络中心度特征(简化版)\"\"\"
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 出度和入度
    out_degree = tx.groupby('src')['dst'].nunique().reset_index()
    out_degree = out_degree.rename(columns={'src': 'wallet_id', 'dst': 'out_degree'})
    
    in_degree = tx.groupby('dst')['src'].nunique().reset_index()
    in_degree = in_degree.rename(columns={'dst': 'wallet_id', 'src': 'in_degree'})
    
    # 交易次数
    out_cnt = tx.groupby('src').size().reset_index(name='out_cnt')
    out_cnt = out_cnt.rename(columns={'src': 'wallet_id'})
    in_cnt = tx.groupby('dst').size().reset_index(name='in_cnt')
    in_cnt = in_cnt.rename(columns={'dst': 'wallet_id'})
    
    # 合并
    result = out_degree.merge(in_degree, on='wallet_id', how='outer').fillna(0)
    result = result.merge(out_cnt, on='wallet_id', how='outer').fillna(0)
    result = result.merge(in_cnt, on='wallet_id', how='outer').fillna(0)
    
    result['total_degree'] = result['out_degree'] + result['in_degree']
    result['out_strength'] = result['out_cnt'] / (result['out_degree'] + 1)
    result['in_strength'] = result['in_cnt'] / (result['in_degree'] + 1)
    result['strength_max'] = result[['out_strength', 'in_strength']].max(axis=1)
    
    return result[['wallet_id', 'total_degree', 'strength_max']]

result = feature_wallet_centrality_simple(open_df, tx_df, close_df)
