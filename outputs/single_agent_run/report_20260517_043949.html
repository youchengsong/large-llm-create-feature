def feature_tx_night_ratio(open_df, tx_df, close_df):
    """夜间交易占比特征：凌晨0-5时交易占总交易的比例"""
    import pandas as pd
    tx = tx_df.copy()
    tx['hour'] = pd.to_datetime(tx['timestamp']).dt.hour
    
    # 按源钱包统计
    src_stats = tx.groupby('src').agg(
        total_tx=('tx_id', 'count'),
        night_tx=('hour', lambda x: ((x >= 0) & (x < 5)).sum())
    ).reset_index()
    src_stats.columns = ['wallet_id', 'total_tx_src', 'night_tx_src']
    src_stats['night_ratio_src'] = src_stats['night_tx_src'] / src_stats['total_tx_src'].clip(lower=1)
    
    # 按目标钱包统计
    dst_stats = tx.groupby('dst').agg(
        total_tx=('tx_id', 'count'),
        night_tx=('hour', lambda x: ((x >= 0) & (x < 5)).sum())
    ).reset_index()
    dst_stats.columns = ['wallet_id', 'total_tx_dst', 'night_tx_dst']
    dst_stats['night_ratio_dst'] = dst_stats['night_tx_dst'] / dst_stats['total_tx_dst'].clip(lower=1)
    
    # 合并
    result = pd.merge(src_stats, dst_stats, on='wallet_id', how='outer').fillna(0)
    result['night_ratio'] = (result['night_ratio_src'] + result['night_ratio_dst']) / 2
    result['night_ratio'] = result['night_ratio'].fillna(0)
    return result[['wallet_id', 'night_ratio']]

tx_amount_cv — 交易金额的变异系数（标准差/均值）。洗钱交易中，金额往往有规律性（固定金额或固定 
def feature_tx_amount_cv(open_df, tx_df, close_df):
    """交易金额变异系数：金额标准差/均值，衡量金额波动程度"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 按源钱包统计
    src_stats = tx.groupby('src')['amount'].agg(['mean', 'std', 'count']).reset_index()
    src_stats.columns = ['wallet_id', 'mean_src', 'std_src', 'count_src']
    src_stats['cv_src'] = src_stats['std_src'] / src_stats['mean_src'].clip(lower=0.01)
    src_stats['cv_src'] = src_stats['cv_src'].fillna(0).replace([np.inf, -np.inf], 0)
    
    # 按目标钱包统计
    dst_stats = tx.groupby('dst')['amount'].agg(['mean', 'std']).reset_index()
    dst_stats.columns = ['wallet_id', 'mean_dst', 'std_dst']
    dst_stats['cv_dst'] = dst_stats['std_dst'] / dst_stats['mean_dst'].clip(lower=0.01)
    dst_stats['cv_dst'] = dst_stats['cv_dst'].fillna(0).replace([np.inf, -np.inf], 0)
    
    result = pd.merge(src_stats[['wallet_id', 'cv_src']], dst_stats[['wallet_id', 'cv_dst']], on='wallet_id', how='outer').fillna(0)
    result['amount_cv'] = (result['cv_src'] + result['cv_dst']) / 2
    
    return result[['wallet_id', 'amount_cv']]

推理: 风险指标中提到"交易金额波动幅度"和"交易金额历史均值：交易金额波动剧烈，与历史均值偏离过大"。金额变异系数可以量化这一特征，区分规律性洗钱和正常随机交易。

tx_peak_hour_concentration_v2 — 交易时段集中度（优化版）：钱包最活跃的3个小时的交易占总交易的比例。洗钱操作通常 
def feature_tx_peak_hour_concentration_v2(open_df, tx_df, close_df):
    """交易时段集中度：计算钱包交易最集中的3个小时的交易占比（优化版）"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    tx['hour'] = pd.to_datetime(tx['timestamp']).dt.hour
    
    # 按源钱包统计各小时交易数 - 使用pivot_table更高效
    src_hourly = tx.groupby(['src', 'hour']).size().reset_index(name='count')
    
    # 计算每个钱包的总交易数和top3小时
    src_total = src_hourly.groupby('src')['count'].sum().reset_index(name='total')
    
    # 取每个钱包top3小时
    src_top3 = src_hourly.sort_values(['src', 'count'], ascending=[True, False])
    src_top3 = src_top3.groupby('src').head(3)
    src_top3_sum = src_top3.groupby('src')['count'].sum().reset_index(name='top3_sum')
    
    src_conc = src_total.merge(src_top3_sum, on='src', how='left')
    src_conc['concentration_src'] = src_conc['top3_sum'] / src_conc['total'].clip(lower=1)
    src_conc = src_conc[['src', 'concentration_src']].rename(columns={'src': 'wallet_id'})
    
    # 按目标钱包
    dst_hourly = tx.groupby(['dst', 'hour']).size().reset_index(name='count')
    dst_total = dst_hourly.groupby('dst')['count'].sum().reset_index(name='total')
    dst_top3 = dst_hourly.sort_values(['dst', 'count'], ascending=[True, False])
    dst_top3 = dst_top3.groupby('dst').head(3)
    dst_top3_sum = dst_top3.groupby('dst')['count'].sum().reset_index(name='top3_sum')
    
    dst_conc = dst_total.merge(dst_top3_sum, on='dst', how='left')
    dst_conc['concentration_dst'] = dst_conc['top3_sum'] / dst_conc['total'].clip(lower=1)
    dst_conc = dst_conc[['dst', 'concentration_dst']].rename(columns={'dst': 'wallet_id'})
    
    result = pd.merge(src_conc, dst_conc, on='wallet_id', how='outer').fillna(0)
    result['peak_hour_concentration'] = (result['concentration_src'] + result['concentration_dst']) / 2
    
    return result[['wallet_id', 'peak_hour_concentration']]

tx_net_amount_flow — 资金净流量比率（转出总额/转入总额）。洗钱链路中，中间节点呈现"快进快出"的平衡 
def feature_tx_net_amount_flow(open_df, tx_df, close_df):
    """资金净流量特征：钱包转出总额与转入总额的比值"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 作为源：转出总额
    src_out = tx.groupby('src')['amount'].sum().reset_index()
    src_out.columns = ['wallet_id', 'total_out_amount']
    
    # 作为目标：转入总额
    dst_in = tx.groupby('dst')['amount'].sum().reset_index()
    dst_in.columns = ['wallet_id', 'total_in_amount']
    
    result = pd.merge(src_out, dst_in, on='wallet_id', how='outer').fillna(0)
    
    # 净流量比率：转出/转入。>>1表示净转出（资金流出），<<1表示净转入（资金汇聚）
    result['net_flow_ratio'] = result['total_out_amount'] / result['total_in_amount'].clip(lower=0.01)
    result['net_flow_ratio'] = result['net_flow_ratio'].replace([np.inf, -np.inf], 100).clip(0, 100)
    
    # 净流量方向：1=净转出, -1=净转入, 0=平衡
    result['net_flow_direction'] = np.where(
        result['net_flow_ratio'] > 2, 1,
        np.where(result['net_flow_ratio'] < 0.5, -1, 0)
    )
    
    return result[['wallet_id', 'net_flow_ratio']]

def feature_tx_amount_ratio_to_balance(open_df, tx_df, close_df):
    """交易金额与初始余额的比率：衡量交易金额相对于钱包初始余额的异常程度"""
    import pandas as pd
    import numpy as np
    open_df = open_df.copy()
    tx = tx_df.copy()
    
    # 合并初始余额
    wallet_balance = open_df[['wallet_id', 'init_balance']].copy()
    
    # 按源钱包统计平均转出金额
    src_avg = tx.groupby('src')['amount'].mean().reset_index()
    src_avg.columns = ['wallet_id', 'avg_out_amount']
    
    # 按目标钱包统计平均转入金额
    dst_avg = tx.groupby('dst')['amount'].mean().reset_index()
    dst_avg.columns = ['wallet_id', 'avg_in_amount']
    
    result = wallet_balance.merge(src_avg, on='wallet_id', how='left').fillna(0)
    result = result.merge(dst_avg, on='wallet_id', how='left').fillna(0)
    
    # 交易金额与初始余额的比率
    result['amount_balance_ratio'] = (result['avg_out_amount'] + result['avg_in_amount']) / 2 / result['init_balance'].clip(lower=1)
    result['amount_balance_ratio'] = result['amount_balance_ratio'].clip(0, 1000)
    
    return result[['wallet_id', 'amount_balance_ratio']]

def feature_tx_amount_precision_anomaly(open_df, tx_df, close_df):

    import numpy as np

    # 按源钱包统计

        total_src=('tx_id', 'count'),
        cents_src=('has_cents', 'sum')
    ).reset_index()

    src_stats['cents_ratio_src'] = src_stats['cents_src_tx'] / src_stats['total_src_tx'].clip(lower=1)
    
    # 按目标钱包统计
    dst_stats = tx.groupby('dst').agg(
        total_dst=('tx_id', 'count'),
        cents_dst=('has_cents', 'sum')
    ).reset_index()
    dst_stats.columns = ['wallet_id', 'total_dst_tx', 'cents_dst_tx']
    dst_stats['cents_ratio_dst'] = dst_stats['cents_dst_tx'] / dst_stats['total_dst_tx'].clip(lower=1)

                      dst_stats[['wallet_id', 'cents_ratio_dst']],

    result['amount_precision_anomaly'] = 1 - ((result['cents_ratio_src'] + result['cents_ratio_dst']) / 2)
    
    return result[['wallet_id', 'amount_precision_anomaly']]

tx_counterparty_repeat_rate — 交易对手重复率：钱包与同一对手的最大交易次数占比。洗钱链路中，中间节点与固定对手 
def feature_tx_counterparty_repeat_rate(open_df, tx_df, close_df):
    """交易对手重复率：与同一对手交易次数占总次数的比例（最大重复率）"""
    import pandas as pd
    tx = tx_df.copy()
    
    # 按源钱包统计：与每个对手的交易次数，取最大占比
    def max_repeat_rate(group):
        counts = group.value_counts()
        if len(counts) <= 1:
            return 0.0
        return counts.iloc[0] / counts.sum()
    
    src_repeat = tx.groupby('src')['dst'].apply(max_repeat_rate).reset_index()
    src_repeat.columns = ['wallet_id', 'repeat_rate_src']
    
    dst_repeat = tx.groupby('dst')['src'].apply(max_repeat_rate).reset_index()
    dst_repeat.columns = ['wallet_id', 'repeat_rate_dst']
    
    result = pd.merge(src_repeat, dst_repeat, on='wallet_id', how='outer').fillna(0)
    result['counterparty_repeat_rate'] = (result['repeat_rate_src'] + result['repeat_rate_dst']) / 2
    
    return result[['wallet_id', 'counterparty_repeat_rate']]

推理: 裁判文书提到"交易对手固定"是线下跑分模式的特征，以及"固定上线指定收款账户"的操作模式。与固定对手的高频重复交易是洗钱网络的重要特征。

def feature_tx_amount_near_limit_ratio(open_df, tx_df, close_df):
    """接近限额交易比例：金额接近常见限额（2000/5000/10000等）的交易占比"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 常见限额：2000, 5000, 10000, 20000, 50000
    limits = [2000, 5000, 10000, 20000, 50000]
    
    def is_near_limit(amount):
        return any(abs(amount - limit) / limit < 0.05 for limit in limits)
    
    tx['near_limit'] = tx['amount'].apply(is_near_limit)
    
    # 按源钱包统计
    src_stats = tx.groupby('src').agg(
        total_src=('tx_id', 'count'),
        near_limit_src=('near_limit', 'sum')
    ).reset_index()
    src_stats.columns = ['wallet_id', 'total_src_tx', 'near_limit_src_tx']
    src_stats['near_limit_ratio_src'] = src_stats['near_limit_src_tx'] / src_stats['total_src_tx'].clip(lower=1)
    
    # 按目标钱包统计
    dst_stats = tx.groupby('dst').agg(
        total_dst=('tx_id', 'count'),
        near_limit_dst=('near_limit', 'sum')
    ).reset_index()
    dst_stats.columns = ['wallet_id', 'total_dst_tx', 'near_limit_dst_tx']
    dst_stats['near_limit_ratio_dst'] = dst_stats['near_limit_dst_tx'] / dst_stats['total_dst_tx'].clip(lower=1)
    
    result = pd.merge(src_stats[['wallet_id', 'near_limit_ratio_src']], 
                      dst_stats[['wallet_id', 'near_limit_ratio_dst']], 
                      on='wallet_id', how='outer').fillna(0)
    result['near_limit_ratio'] = (result['near_limit_ratio_src'] + result['near_limit_ratio_dst']) / 2
    
    return result[['wallet_id', 'near_limit_ratio']]

推理: 交易模式说明中"四类钱包洗钱"明确提到"存在明显的日累计额度控制特征"。洗钱分子会尽量接近限额以最大化资金流转效率，导致金额接近常见限额。

def feature_tx_device_count_ratio(open_df, tx_df, close_df):
    """设备-钱包比率：钱包使用的设备数与其交易次数的比值"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 按源钱包：交易次数和唯一设备数
    src_stats = tx.groupby('src').agg(
        tx_count=('tx_id', 'count'),
        device_count=('src_device', 'nunique')
    ).reset_index()
    src_stats.columns = ['wallet_id', 'tx_count_src', 'device_count_src']
    src_stats['device_tx_ratio_src'] = src_stats['device_count_src'] / src_stats['tx_count_src'].clip(lower=1)
    
    # 按目标钱包
    dst_stats = tx.groupby('dst').agg(
        tx_count=('tx_id', 'count'),
        device_count=('dst_device', 'nunique')
    ).reset_index()
    dst_stats.columns = ['wallet_id', 'tx_count_dst', 'device_count_dst']
    dst_stats['device_tx_ratio_dst'] = dst_stats['device_count_dst'] / dst_stats['tx_count_dst'].clip(lower=1)
    
    result = pd.merge(src_stats[['wallet_id', 'device_tx_ratio_src', 'device_count_src']], 
                      dst_stats[['wallet_id', 'device_tx_ratio_dst', 'device_count_dst']], 
                      on='wallet_id', how='outer').fillna(0)
    
    # 取较大值：设备数/交易数比率高表示频繁换设备
    result['device_count_ratio'] = result[['device_tx_ratio_src', 'device_tx_ratio_dst']].max(axis=1)
    # 也保留设备总数
    result['total_device_count'] = result['device_count_src'] + result['device_count_dst']
    
    return result[['wallet_id', 'device_count_ratio']]

推理: 裁判文书提到"操作员负责手机转账操作"和"多卡轮换操作"，涉及设备使用模式。设备与交易的比率异常可以反映洗钱操作的组织化特征。
