def feature_night_transaction_ratio(open_df, tx_df, close_df):
    """凌晨交易占比：检测凌晨0-5时交易比例，对应线上跑分/夜班操作模式"""
    import pandas as pd
    tx = tx_df.copy()

    tx['is_night'] = tx['hour'].between(0, 5).astype(int)
    src_night = tx.groupby('src')['is_night'].agg(['mean', 'sum', 'count']).reset_index()

    src_night['night_tx_count'] = src_night['night_count']
    result = src_night[['wallet_id', 'night_transaction_ratio', 'night_tx_count']]
    result = result.fillna(0)
    return result

def feature_rapid_in_out_ratio(open_df, tx_df, close_df):
    """即转即取特征：检测钱包在收到资金后短时间内转出的比例"""
    import pandas as pd
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按源钱包排序，计算每笔交易与上一笔的时间间隔
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    tx_sorted['prev_timestamp'] = tx_sorted.groupby('src')['timestamp'].shift(1)
    tx_sorted['interval_minutes'] = (tx_sorted['timestamp'] - tx_sorted['prev_timestamp']).dt.total_seconds() / 60.0
    
    # 极短间隔：间隔小于5分钟
    tx_sorted['is_rapid'] = (tx_sorted['interval_minutes'] < 5).astype(int)
    
    # 统计每个钱包的快速交易比例
    rapid_stats = tx_sorted.groupby('src')['is_rapid'].agg(['mean', 'sum', 'count']).reset_index()
    rapid_stats.columns = ['wallet_id', 'rapid_ratio', 'rapid_count', 'total_count']
    
    # 计算中位数间隔
    median_interval = tx_sorted.groupby('src')['interval_minutes'].median().reset_index()
    median_interval.columns = ['wallet_id', 'median_interval_minutes']
    
    result = rapid_stats.merge(median_interval, on='wallet_id', how='outer')
    result = result.fillna(0)
    return result

result = feature_rapid_in_out_ratio(open_df, tx_df, close_df)

amount_volatility_pattern — 计算交易金额的变异系数、跨度比、最大单笔占比等波动性指标。洗钱活动常表现为金额分 
def feature_amount_volatility_pattern(open_df, tx_df, close_df):
    """金额波动模式：检测交易金额的波动性、集中度和异常模式"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 按源钱包统计金额特征
    src_amount = tx.groupby('src')['amount'].agg(
        amount_std='std',
        amount_mean='mean',
        amount_max='max',
        amount_min='min',
        amount_sum='sum',
        amount_count='count',
        amount_median='median'
    ).reset_index()
    src_amount.columns = ['wallet_id', 'amount_std', 'amount_mean', 'amount_max', 
                          'amount_min', 'amount_sum', 'amount_count', 'amount_median']
    src_amount = src_amount.fillna(0)
    
    # 变异系数：标准差/均值，衡量波动性
    src_amount['amount_cv'] = src_amount['amount_std'] / src_amount['amount_mean'].clip(lower=0.01)
    
    # 金额跨度：(max-min)/mean
    src_amount['amount_span_ratio'] = (src_amount['amount_max'] - src_amount['amount_min']) / src_amount['amount_mean'].clip(lower=0.01)
    
    # 最大单笔占比
    src_amount['max_amount_ratio'] = src_amount['amount_max'] / src_amount['amount_sum'].clip(lower=0.01)
    
    # 中位数与均值偏离度（偏度指标）
    src_amount['median_mean_ratio'] = src_amount['amount_median'] / src_amount['amount_mean'].clip(lower=0.01)
    
    # 平均每笔金额
    src_amount['avg_amount'] = src_amount['amount_mean']
    
    cols = ['wallet_id', 'amount_cv', 'amount_span_ratio', 'max_amount_ratio', 
            'median_mean_ratio', 'avg_amount', 'amount_sum', 'amount_std']
    return src_amount[cols]

result = feature_amount_volatility_pattern(open_df, tx_df, close_df)

hour_concentration_entropy — 计算交易时间的熵值（分散度）、峰值时段和凌晨活跃度。洗钱活动时间集中度异常——要 
def feature_hour_concentration_entropy(open_df, tx_df, close_df):
    """交易时间集中度与熵值：检测交易时间是否集中在特定时段"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 对每个钱包统计各小时交易分布，计算熵值
    def calc_hour_entropy(group):
        counts = group['hour'].value_counts()
        probs = counts / counts.sum()
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        return entropy
    
    # 按源钱包计算时间熵
    src_entropy = tx.groupby('src').apply(calc_hour_entropy).reset_index()
    src_entropy.columns = ['wallet_id', 'hour_entropy']
    
    # 峰值小时交易占比
    hour_peak = tx.groupby(['src', 'hour']).size().reset_index()
    hour_peak.columns = ['wallet_id', 'hour', 'count']
    hour_peak = hour_peak.sort_values(['wallet_id', 'count'], ascending=[True, False])
    hour_peak_max = hour_peak.groupby('wallet_id').first().reset_index()
    hour_peak_max.columns = ['wallet_id', 'peak_hour', 'peak_count']
    
    hour_total = tx.groupby('src').size().reset_index()
    hour_total.columns = ['wallet_id', 'total_tx']
    
    result = hour_peak_max.merge(hour_total, on='wallet_id', how='outer')
    result['peak_hour_ratio'] = result['peak_count'] / result['total_tx'].clip(lower=1)
    
    # 合并熵值
    result = result.merge(src_entropy, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 凌晨活跃度（0-5时的交易占比）
    tx['is_early_morning'] = tx['hour'].between(0, 5).astype(int)
    early_morning = tx.groupby('src')['is_early_morning'].mean().reset_index()
    early_morning.columns = ['wallet_id', 'early_morning_ratio']
    
    result = result.merge(early_morning, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    return result[['wallet_id', 'hour_entropy', 'peak_hour', 'peak_hour_ratio', 'early_morning_ratio']]

result = feature_hour_concentration_entropy(open_df, tx_df, close_df)

推理: 裁判文书指出"犯罪操作集中在下午(14-17时)最多"和"夜班集中操作（22:00-04:00）"。交易模式也提到洗钱"主要发生在深夜或低活跃时段"。时间熵值能量化这种集中度异常。

amount_percentile_and_tail — 检测钱包交易金额在全局分布中的位置特征：高百分位（大额）交易占比、低百分位（小额 
def feature_amount_percentile_and_tail(open_df, tx_df, close_df):
    """金额百分位与尾部特征：检测钱包交易金额在整个数据集中的相对位置"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 计算全局金额分布
    global_amounts = tx['amount'].values
    p25, p50, p75, p90, p95, p99 = np.percentile(global_amounts, [25, 50, 75, 90, 95, 99])
    
    # 对每个钱包计算其交易金额在全局分布中的位置
    def calc_percentile_features(group):
        amounts = group['amount'].values
        if len(amounts) == 0:
            return pd.Series({'above_p90_ratio': 0, 'above_p95_ratio': 0, 
                             'above_p99_ratio': 0, 'below_p25_ratio': 0})
        
        above_p90 = np.sum(amounts > p90) / len(amounts)
        above_p95 = np.sum(amounts > p95) / len(amounts)
        above_p99 = np.sum(amounts > p99) / len(amounts)
        below_p25 = np.sum(amounts < p25) / len(amounts)
        
        return pd.Series({'above_p90_ratio': above_p90, 'above_p95_ratio': above_p95, 
                         'above_p99_ratio': above_p99, 'below_p25_ratio': below_p25})
    
    # 按源钱包计算
    src_percentiles = tx.groupby('src').apply(calc_percentile_features).reset_index()
    
    # 按目标钱包计算
    dst_percentiles = tx.groupby('dst').apply(calc_percentile_features).reset_index()
    dst_percentiles.columns = ['wallet_id', 'dst_above_p90_ratio', 'dst_above_p95_ratio', 
                               'dst_above_p99_ratio', 'dst_below_p25_ratio']
    
    result = src_percentiles.merge(dst_percentiles, left_on='src', right_on='wallet_id', how='outer')
    result = result.drop('src', axis=1, errors='ignore')
    result = result.fillna(0)
    
    # 极端金额偏离：最大金额/中位数
    src_amount = tx.groupby('src')['amount'].agg(['max', 'median', 'mean']).reset_index()
    src_amount.columns = ['wallet_id', 'max_amount', 'median_amount', 'mean_amount']
    src_amount['max_median_ratio'] = src_amount['max_amount'] / src_amount['median_amount'].clip(lower=0.01)
    src_amount['max_mean_ratio'] = src_amount['max_amount'] / src_amount['mean_amount'].clip(lower=0.01)
    
    result = result.merge(src_amount[['wallet_id', 'max_median_ratio', 'max_mean_ratio']], 
                          on='wallet_id', how='outer')
    result = result.fillna(0)
    
    return result

result = feature_amount_percentile_and_tail(open_df, tx_df, close_df)

weekend_holiday_tx_ratio — 检测钱包在周末、非工作时间、晚间和凌晨的交易比例。洗钱活动常发生在非工作时间（凌 
def feature_weekend_holiday_tx_ratio(open_df, tx_df, close_df):
    """周末/节假日交易占比：检测钱包在周末和非工作时间的交易比例"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 星期几（0=周一，6=周日）
    tx['day_of_week'] = tx['timestamp'].dt.dayofweek
    # 是否为周末（周六日）
    tx['is_weekend'] = (tx['day_of_week'] >= 5).astype(int)
    # 是否为工作时间（9-17点）
    tx['is_work_hour'] = tx['hour'].between(9, 17).astype(int)
    # 是否为晚间活跃时段（20-23点）
    tx['is_evening'] = tx['hour'].between(20, 23).astype(int)
    # 是否为凌晨时段（0-5点）
    tx['is_dawn'] = tx['hour'].between(0, 5).astype(int)
    
    # 按源钱包统计
    src_time_pattern = tx.groupby('src').agg(
        weekend_ratio=('is_weekend', 'mean'),
        work_hour_ratio=('is_work_hour', 'mean'),
        evening_ratio=('is_evening', 'mean'),
        dawn_ratio=('is_dawn', 'mean'),
        weekend_count=('is_weekend', 'sum'),
        dawn_count=('is_dawn', 'sum')
    ).reset_index()
    src_time_pattern.columns = ['wallet_id', 'weekend_tx_ratio', 'work_hour_tx_ratio',
                                 'evening_tx_ratio', 'dawn_tx_ratio',
                                 'weekend_tx_count', 'dawn_tx_count']
    
    # 非工作时间交易占比（周末+晚间+凌晨）
    src_time_pattern['non_work_hour_ratio'] = 1 - src_time_pattern['work_hour_tx_ratio']
    src_time_pattern['abnormal_hour_ratio'] = src_time_pattern['dawn_tx_ratio'] + src_time_pattern['evening_tx_ratio']
    
    result = src_time_pattern.fillna(0)
    
    cols = ['wallet_id', 'weekend_tx_ratio', 'work_hour_tx_ratio', 'evening_tx_ratio',
            'dawn_tx_ratio', 'non_work_hour_ratio', 'abnormal_hour_ratio']
    return result[cols]

result = feature_weekend_holiday_tx_ratio(open_df, tx_df, close_df)

amount_sequence_trend — 检测交易金额的时序变化趋势：金额递增/递减/稳定的比例、方向一致性。洗钱行为常呈 
def feature_amount_sequence_trend(open_df, tx_df, close_df):
    """交易金额序列趋势：检测钱包交易金额的时序变化趋势和模式"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    
    # 按源钱包和时间排序
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    # 计算每笔交易金额的变化
    tx_sorted['prev_amount'] = tx_sorted.groupby('src')['amount'].shift(1)
    tx_sorted['amount_change'] = tx_sorted['amount'] - tx_sorted['prev_amount']
    tx_sorted['amount_change_pct'] = tx_sorted['amount_change'] / tx_sorted['prev_amount'].clip(lower=0.01)
    tx_sorted['amount_direction'] = np.sign(tx_sorted['amount_change'])
    
    # 金额上升趋势比例（金额递增的交易占比）
    tx_sorted['is_increasing'] = (tx_sorted['amount_change'] > 0).astype(int)
    tx_sorted['is_decreasing'] = (tx_sorted['amount_change'] < 0).astype(int)
    tx_sorted['is_stable'] = (tx_sorted['amount_change'] == 0).astype(int)
    
    # 去掉第一笔（prev_amount为NaN）
    tx_with_prev = tx_sorted[tx_sorted['prev_amount'].notna()].copy()
    
    # 按钱包统计
    trend_stats = tx_with_prev.groupby('src').agg(
        increasing_ratio=('is_increasing', 'mean'),
        decreasing_ratio=('is_decreasing', 'mean'),
        stable_ratio=('is_stable', 'mean'),
        avg_abs_change=('amount_change', lambda x: np.abs(x).mean()),
        avg_change_pct=('amount_change_pct', 'mean'),
        change_std=('amount_change', 'std'),
        positive_change_ratio=('amount_direction', lambda x: np.mean(x > 0)),
        negative_change_ratio=('amount_direction', lambda x: np.mean(x < 0))
    ).reset_index()
    trend_stats.columns = ['wallet_id', 'increasing_ratio', 'decreasing_ratio', 'stable_ratio',
                            'avg_abs_change', 'avg_change_pct', 'change_std',
                            'positive_change_ratio', 'negative_change_ratio']
    
    # 金额变化的方向一致性（连续同向变化的比例）
    tx_with_prev['same_direction'] = (tx_with_prev['amount_direction'] == 
                                       tx_with_prev.groupby('src')['amount_direction'].shift(1)).astype(int)
    direction_consistency = tx_with_prev.groupby('src')['same_direction'].mean().reset_index()
    direction_consistency.columns = ['wallet_id', 'direction_consistency']
    
    result = trend_stats.merge(direction_consistency, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    return result

result = feature_amount_sequence_trend(open_df, tx_df, close_df)

推理: 裁判文书提到洗钱操作员批量转账时使用固定金额或固定比例扣除佣金后的金额。金额序列的规律性是机器批量操作的典型特征，区别于正常用户的随机消费行为。

dst_diversity_and_concentration — 检测钱包交易对手的集中度：唯一交易对手数、最大交易对手占比、前3大占比、赫芬达尔 
def feature_dst_diversity_and_concentration(open_df, tx_df, close_df):
    """交易对手集中度：检测钱包的交易对手集中度和重复交易比例"""
    import pandas as pd
    import numpy as np
    tx = tx_df.copy()
    
    # 使用向量化方法计算交易对手集中度
    
    # 每个源钱包的交易对手数量
    n_unique_dst = tx.groupby('src')['dst'].nunique().reset_index()
    n_unique_dst.columns = ['wallet_id', 'n_unique_dst']
    
    # 每个源钱包的总交易数
    tx_count = tx.groupby('src').size().reset_index()
    tx_count.columns = ['wallet_id', 'total_tx']
    
    # 每个源钱包-目标对的交易次数
    pair_counts = tx.groupby(['src', 'dst']).size().reset_index()
    pair_counts.columns = ['wallet_id', 'dst', 'pair_count']
    
    # 计算每个源钱包的最大交易对手占比
    max_pair = pair_counts.groupby('wallet_id')['pair_count'].max().reset_index()
    max_pair.columns = ['wallet_id', 'max_pair_count']
    
    # 前3大交易对手占比
    pair_counts_sorted = pair_counts.sort_values(['wallet_id', 'pair_count'], ascending=[True, False])
    top3 = pair_counts_sorted.groupby('wallet_id').head(3).groupby('wallet_id')['pair_count'].sum().reset_index()
    top3.columns = ['wallet_id', 'top3_pair_sum']
    
    # 合并
    result = n_unique_dst.merge(tx_count, on='wallet_id', how='outer')
    result = result.merge(max_pair, on='wallet_id', how='outer')
    result = result.merge(top3, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    result['top1_dst_ratio'] = result['max_pair_count'] / result['total_tx'].clip(lower=1)
    result['top3_dst_ratio'] = result['top3_pair_sum'] / result['total_tx'].clip(lower=1)
    
    # 赫芬达尔指数
    pair_counts['pair_prob'] = pair_counts['pair_count'] / pair_counts.groupby('wallet_id')['pair_count'].transform('sum')
    pair_counts['pair_prob_sq'] = pair_counts['pair_prob'] ** 2
    hhi = pair_counts.groupby('wallet_id')['pair_prob_sq'].sum().reset_index()
    hhi.columns = ['wallet_id', 'herfindahl_index']
    
    result = result.merge(hhi, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 目标钱包的源集中度
    n_unique_src = tx.groupby('dst')['src'].nunique().reset_index()
    n_unique_src.columns = ['wallet_id', 'n_unique_src']
    
    result = result.merge(n_unique_src, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 总交易对手数
    result['total_counterparties'] = result['n_unique_dst'] + result['n_unique_src']
    
    cols = ['wallet_id', 'n_unique_dst', 'n_unique_src', 'total_counterparties',
            'top1_dst_ratio', 'top3_dst_ratio', 'herfindahl_index']
    return result[cols]

result = feature_dst_diversity_and_concentration(open_df, tx_df, close_df)

推理: 交易模式中洗钱行为包含"多对一、一对多"模式。赫芬达尔指数能有效量化交易对手的集中/分散程度。
