amount_roundness_score — 计算交易金额的"整度"评分。洗钱交易金额常为整数（如1000、5000等）或接近 
def feature_amount_roundness_score(open_df, tx_df, close_df):
    """
    金额"整度"评分：洗钱交易中金额常接近限额或为整数倍，
    正常消费金额更随机（如1.23元而非100元整）
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 对每笔交易计算金额的"整度"指标
    # 1. 是否为整数（无小数）
    tx['is_round'] = (tx['amount'] == tx['amount'].astype(int)).astype(int)
    
    # 2. 金额是否为100的倍数
    tx['is_multiple_of_100'] = (tx['amount'] % 100 == 0).astype(int)
    
    # 3. 金额是否为1000的倍数
    tx['is_multiple_of_1000'] = (tx['amount'] % 1000 == 0).astype(int)
    
    # 4. 金额是否接近限额
    limits = [5000, 10000, 20000, 50000]
    tx['near_limit'] = 0
    for limit in limits:
        tx['near_limit'] = tx['near_limit'] | (np.abs(tx['amount'] - limit) / limit < 0.05).astype(int)
    
    # 计算每个钱包的平均整度得分
    tx['roundness_score'] = (tx['is_round'] * 0.3 + tx['is_multiple_of_100'] * 0.3 + 
                             tx['is_multiple_of_1000'] * 0.2 + tx['near_limit'] * 0.2)
    
    # 按源钱包汇总
    src_features = tx.groupby('src').agg(
        src_tx_count=('tx_id', 'count'),
        src_roundness_mean=('roundness_score', 'mean'),
        src_roundness_std=('roundness_score', 'std'),
        src_round_tx_ratio=('is_round', 'mean'),
        src_limit_tx_ratio=('near_limit', 'mean'),
        src_multiple_100_ratio=('is_multiple_of_100', 'mean'),
    ).reset_index()
    src_features.columns = ['wallet_id', 'tx_count_src', 'roundness_mean_src', 'roundness_std_src', 
                            'round_tx_ratio_src', 'limit_tx_ratio_src', 'multiple_100_ratio_src']
    
    # 按目标钱包汇总
    dst_features = tx.groupby('dst').agg(
        dst_tx_count=('tx_id', 'count'),
        dst_roundness_mean=('roundness_score', 'mean'),
        dst_round_tx_ratio=('is_round', 'mean'),
    ).reset_index()
    dst_features.columns = ['wallet_id', 'tx_count_dst', 'roundness_mean_dst', 'round_tx_ratio_dst']
    
    # 合并
    result = pd.merge(src_features, dst_features, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 综合得分
    result['amount_roundness_score'] = (result['roundness_mean_src'] + result['roundness_mean_dst']) / 2
    result['amount_roundness_max'] = result[['roundness_mean_src', 'roundness_mean_dst']].max(axis=1)
    
    return result[['wallet_id', 'amount_roundness_score', 'amount_roundness_max', 
                    'round_tx_ratio_src', 'limit_tx_ratio_src', 'multiple_100_ratio_src']]

result = feature_amount_roundness_score(open_df, tx_df, close_df)

def feature_amount_concentration_entropy(open_df, tx_df, close_df):
    """
    交易金额集中度/熵值：洗钱交易金额集中在特定区间（如限额附近），
    而正常交易金额分布更分散。用熵值衡量金额多样性。
    """
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    
    # 将金额分桶（按10元区间）
    tx['amount_bucket'] = (tx['amount'] // 10).astype(int)
    
    # 按源钱包计算金额分布熵
    def calc_entropy(group):
        counts = group.value_counts()
        probs = counts / counts.sum()
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        return entropy
    
    # 源钱包维度
    src_entropy = tx.groupby('src')['amount_bucket'].apply(calc_entropy).reset_index()
    src_entropy.columns = ['wallet_id', 'amount_entropy_src']
    
    # 源钱包金额集中度（top3金额占比）
    def top3_concentration(group):
        counts = group.value_counts()
        if len(counts) == 0:
            return 0.0
        top3_sum = counts.head(3).sum()
        return top3_sum / counts.sum()
    
    src_concentration = tx.groupby('src')['amount_bucket'].apply(top3_concentration).reset_index()
    src_concentration.columns = ['wallet_id', 'amount_concentration_src']
    
    # 源钱包金额的变异系数
    src_cv = tx.groupby('src')['amount'].agg(['mean', 'std']).reset_index()
    src_cv.columns = ['wallet_id', 'amount_mean_src', 'amount_std_src']
    src_cv['amount_cv_src'] = src_cv['amount_std_src'] / (src_cv['amount_mean_src'] + 1e-10)
    
    # 目标钱包维度
    dst_entropy = tx.groupby('dst')['amount_bucket'].apply(calc_entropy).reset_index()
    dst_entropy.columns = ['wallet_id', 'amount_entropy_dst']
    
    dst_concentration = tx.groupby('dst')['amount_bucket'].apply(top3_concentration).reset_index()
    dst_concentration.columns = ['wallet_id', 'amount_concentration_dst']
    
    # 合并所有特征
    result = src_entropy.merge(src_concentration, on='wallet_id', how='outer')
    result = result.merge(src_cv, on='wallet_id', how='outer')
    result = result.merge(dst_entropy, on='wallet_id', how='outer')
    result = result.merge(dst_concentration, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 综合指标：低熵+高集中度 = 异常
    result['amount_anomaly_score'] = (1 - result['amount_entropy_src'] / 10) * result['amount_concentration_src']
    
    return result[['wallet_id', 'amount_entropy_src', 'amount_concentration_src', 
                    'amount_cv_src', 'amount_entropy_dst', 'amount_concentration_dst',
                    'amount_anomaly_score']]

result = feature_amount_concentration_entropy(open_df, tx_df, close_df)

推理: 根据裁判文书，跑分交易金额有固定模式（如按佣金比例转出=转入×(1-佣金率)），金额集中在特定值附近。正常消费金额分布更随机多样。

def feature_night_tx_ratio_and_volume(open_df, tx_df, close_df):
    """
    夜间交易比率与金额占比：洗钱交易集中在深夜/凌晨时段（22:00-04:00），
    正常用户在该时段交易极少。计算夜间交易数量和金额占比。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 定义夜间时段：22:00-04:00（深夜跑分时段）
    tx['is_night'] = ((tx['hour'] >= 22) | (tx['hour'] <= 4)).astype(int)
    
    # 凌晨时段：0:00-5:00（在线洗钱模式）
    tx['is_dawn'] = (tx['hour'] <= 5).astype(int)
    
    # 下午时段：14:00-17:00（犯罪操作高峰）
    tx['is_afternoon'] = ((tx['hour'] >= 14) & (tx['hour'] <= 17)).astype(int)
    
    # 按源钱包汇总
    src_features = tx.groupby('src').agg(
        tx_count=('tx_id', 'count'),
        night_tx_count=('is_night', 'sum'),
        night_tx_ratio=('is_night', 'mean'),
        night_amount_sum=('amount', lambda x: x[tx.loc[x.index, 'is_night'] == 1].sum()),
        dawn_tx_ratio=('is_dawn', 'mean'),
        afternoon_tx_ratio=('is_afternoon', 'mean'),
        total_amount=('amount', 'sum'),
    ).reset_index()
    
    # 计算夜间金额占比
    src_features['night_amount_ratio'] = src_features['night_amount_sum'] / (src_features['total_amount'] + 1)
    
    # 夜间交易密度（每小时夜间交易数）
    src_features['night_tx_density'] = src_features['night_tx_count'] / (src_features['tx_count'] + 1)
    
    src_features.columns = ['wallet_id', 'tx_count', 'night_tx_count', 'night_tx_ratio', 
                            'night_amount_sum', 'dawn_tx_ratio', 'afternoon_tx_ratio',
                            'total_amount', 'night_amount_ratio', 'night_tx_density']
    
    # 按目标钱包汇总
    dst_features = tx.groupby('dst').agg(
        dst_night_tx_ratio=('is_night', 'mean'),
        dst_dawn_tx_ratio=('is_dawn', 'mean'),
    ).reset_index()
    dst_features.columns = ['wallet_id', 'dst_night_tx_ratio', 'dst_dawn_tx_ratio']
    
    result = src_features.merge(dst_features, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    return result[['wallet_id', 'night_tx_ratio', 'night_amount_ratio', 'night_tx_density',
                    'dawn_tx_ratio', 'afternoon_tx_ratio', 'dst_night_tx_ratio', 'dst_dawn_tx_ratio']]

result = feature_night_tx_ratio_and_volume(open_df, tx_df, close_df)

def feature_hour_entropy_and_activity_span(open_df, tx_df, close_df):
    """
    交易时间熵值和活跃时段跨度：洗钱交易的时间分布更均匀（全天候操作）
    或集中在特定异常时段，而正常用户有明确的活跃时段。
    计算时间分布的熵和活跃时段跨度。
    """
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    
    # 按源钱包计算小时分布的熵
    def calc_hour_entropy(group):
        counts = group.value_counts()
        probs = counts / counts.sum()
        entropy = -np.sum(probs * np.log2(probs + 1e-10))
        return entropy
    
    src_hour_entropy = tx.groupby('src')['hour'].apply(calc_hour_entropy).reset_index()
    src_hour_entropy.columns = ['wallet_id', 'hour_entropy_src']
    
    # 活跃时段跨度（最早和最晚交易小时之差）
    def calc_hour_span(group):
        hours = group.values
        if len(hours) < 2:
            return 0
        return max(hours) - min(hours)
    
    src_hour_span = tx.groupby('src')['hour'].apply(calc_hour_span).reset_index()
    src_hour_span.columns = ['wallet_id', 'hour_span_src']
    
    # 交易活跃小时数（有多少不同小时有交易）
    src_active_hours = tx.groupby('src')['hour'].apply(lambda x: len(set(x))).reset_index()
    src_active_hours.columns = ['wallet_id', 'active_hours_src']
    
    # 目标钱包维度
    dst_hour_entropy = tx.groupby('dst')['hour'].apply(calc_hour_entropy).reset_index()
    dst_hour_entropy.columns = ['wallet_id', 'hour_entropy_dst']
    
    dst_hour_span = tx.groupby('dst')['hour'].apply(calc_hour_span).reset_index()
    dst_hour_span.columns = ['wallet_id', 'hour_span_dst']
    
    dst_active_hours = tx.groupby('dst')['hour'].apply(lambda x: len(set(x))).reset_index()
    dst_active_hours.columns = ['wallet_id', 'active_hours_dst']
    
    # 合并
    result = src_hour_entropy.merge(src_hour_span, on='wallet_id', how='outer')
    result = result.merge(src_active_hours, on='wallet_id', how='outer')
    result = result.merge(dst_hour_entropy, on='wallet_id', how='outer')
    result = result.merge(dst_hour_span, on='wallet_id', how='outer')
    result = result.merge(dst_active_hours, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 综合异常分数：高熵值+大跨度 = 全天候操作（异常）
    result['time_anomaly_score'] = result['hour_entropy_src'] * result['hour_span_src'] / 24
    
    return result[['wallet_id', 'hour_entropy_src', 'hour_span_src', 'active_hours_src',
                    'hour_entropy_dst', 'hour_span_dst', 'active_hours_dst', 'time_anomaly_score']]

result = feature_hour_entropy_and_activity_span(open_df, tx_df, close_df)

def feature_amount_hour_profile_deviation(open_df, tx_df, close_df):
    """
    金额-小时画像偏离度：正常用户在不同时段有特定的金额偏好，
    洗钱交易在异常时段（如凌晨）出现大额交易，与正常画像偏离。
    计算每个钱包各时段的平均金额与全局均值的偏离。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    
    # 计算全局每个小时的平均金额（正常画像）
    global_hourly_avg = tx.groupby('hour')['amount'].mean().to_dict()
    
    # 计算每个钱包在每个小时的平均金额
    wallet_hourly = tx.groupby(['src', 'hour'])['amount'].agg(['mean', 'count']).reset_index()
    wallet_hourly.columns = ['wallet_id', 'hour', 'hourly_avg_amount', 'hourly_tx_count']
    
    # 计算与全局画像的偏离度
    wallet_hourly['global_hourly_avg'] = wallet_hourly['hour'].map(global_hourly_avg)
    wallet_hourly['deviation'] = np.abs(wallet_hourly['hourly_avg_amount'] - wallet_hourly['global_hourly_avg']) / (wallet_hourly['global_hourly_avg'] + 1)
    
    # 按钱包汇总偏离度
    deviation_features = wallet_hourly.groupby('wallet_id').agg(
        max_deviation=('deviation', 'max'),
        mean_deviation=('deviation', 'mean'),
        abnormal_hour_count=('deviation', lambda x: (x > 2).sum()),  # 偏离超过2倍的时段数
    ).reset_index()
    
    # 特别关注凌晨时段(0-5点)的金额异常
    dawn_tx = tx[(tx['hour'] >= 0) & (tx['hour'] <= 5)]
    if len(dawn_tx) > 0:
        dawn_amount = dawn_tx.groupby('src')['amount'].agg(['sum', 'mean', 'count']).reset_index()
        dawn_amount.columns = ['wallet_id', 'dawn_amount_sum', 'dawn_amount_mean', 'dawn_tx_count']
    else:
        dawn_amount = pd.DataFrame({'wallet_id': [], 'dawn_amount_sum': [], 'dawn_amount_mean': [], 'dawn_tx_count': []})
    
    result = deviation_features.merge(dawn_amount, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    return result[['wallet_id', 'max_deviation', 'mean_deviation', 'abnormal_hour_count',
                    'dawn_amount_sum', 'dawn_amount_mean', 'dawn_tx_count']]

def feature_consecutive_tx_window(open_df, tx_df, close_df):
    """
    连续交易窗口：洗钱操作员在固定时间窗口内连续操作（如2-8小时完成全部资金流转），
    计算每个钱包的最长连续交易窗口和窗口内的交易密度。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    def calc_tx_windows(group):
        timestamps = group['timestamp'].values
        if len(timestamps) < 3:
            return pd.Series({
                'total_time_span': 0.0,
                'max_window_hours': 0.0,
                'window_tx_density': 0.0,
                'avg_daily_session_count': 0.0
            })
        
        # 总时间跨度（小时）- 转换为秒再计算
        td_total = timestamps[-1] - timestamps[0]
        time_span = td_total / np.timedelta64(1, 'h')
        
        # 检测连续交易窗口（间隔<30分钟视为同一窗口）
        intervals = np.array([timestamps[i] - timestamps[i-1] for i in range(1, len(timestamps))])
        intervals_minutes = intervals / np.timedelta64(1, 'm')
        
        # 窗口分割
        windows = []
        current_window_start = timestamps[0]
        current_window_tx = 1
        
        for i, interval_min in enumerate(intervals_minutes):
            if interval_min < 30:  # 30分钟内视为同一窗口
                current_window_tx += 1
            else:
                window_hours = (timestamps[i] - current_window_start) / np.timedelta64(1, 'h')
                windows.append({'tx_count': current_window_tx, 'hours': float(window_hours)})
                current_window_start = timestamps[i+1]
                current_window_tx = 1
        
        # 最后一个窗口
        window_hours = (timestamps[-1] - current_window_start) / np.timedelta64(1, 'h')
        windows.append({'tx_count': current_window_tx, 'hours': float(window_hours)})
        
        if len(windows) == 0:
            return pd.Series({'total_time_span': float(time_span), 'max_window_hours': 0.0, 
                              'window_tx_density': 0.0, 'avg_daily_session_count': 0.0})
        
        max_window = max(windows, key=lambda w: w['tx_count'])
        
        # 日均会话次数
        days = max(float(time_span) / 24, 1)
        session_count = len(windows)
        
        return pd.Series({
            'total_time_span': float(time_span),
            'max_window_hours': max_window['hours'],
            'window_tx_density': max_window['tx_count'] / (max_window['hours'] + 0.01),
            'avg_daily_session_count': session_count / days
        })
    
    window_features = tx_sorted.groupby('src').apply(calc_tx_windows).reset_index()
    window_features.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result = window_features.fillna(0)
    
    # 异常分数：高窗口密度+长时间跨度 = 集中操作（异常）
    result['consecutive_operation_score'] = result['window_tx_density'] * result['max_window_hours'] / 24
    
    return result[['wallet_id', 'total_time_span', 'max_window_hours', 
                    'window_tx_density', 'avg_daily_session_count', 'consecutive_operation_score']]

tx_amount_acceleration — 计算交易金额序列的变化趋势（加速度）。洗钱交易中金额由小到大逐步增加（测试交易- 
def feature_tx_amount_acceleration(open_df, tx_df, close_df):
    """
    交易金额加速度：洗钱交易中金额由小到大逐步增加（测试交易->大额交易），
    或由大到小快速分散（大额转入->小额转出）。计算金额序列的变化趋势。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    def calc_amount_trend(group):
        amounts = group['amount'].values
        if len(amounts) < 3:
            return pd.Series({
                'amount_trend_slope': 0.0,
                'amount_acceleration': 0.0,
                'amount_first_vs_last': 0.0,
                'amount_increase_ratio': 0.0
            })
        
        # 金额变化斜率（线性回归）
        x = np.arange(len(amounts))
        slope = np.polyfit(x, amounts, 1)[0]
        
        # 二阶差分（加速度）
        first_diff = np.diff(amounts)
        if len(first_diff) > 1:
            acceleration = np.mean(np.diff(first_diff))
        else:
            acceleration = 0
        
        # 首笔与末笔金额比
        first_vs_last = amounts[-1] / (amounts[0] + 1)
        
        # 金额增长比例（最大/最小）
        increase_ratio = np.max(amounts) / (np.min(amounts) + 1)
        
        return pd.Series({
            'amount_trend_slope': slope,
            'amount_acceleration': acceleration,
            'amount_first_vs_last': first_vs_last,
            'amount_increase_ratio': increase_ratio
        })
    
    trend_features = tx_sorted.groupby('src').apply(calc_amount_trend).reset_index()
    trend_features.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result = trend_features.fillna(0)
    
    # 异常分数：高斜率+高加速度 = 金额快速变化（异常）
    # 正斜率=金额递增（测试->大额），负斜率=金额递减（大额->分散）
    result['amount_change_anomaly'] = np.abs(result['amount_trend_slope']) * result['amount_increase_ratio'] / 100
    
    return result[['wallet_id', 'amount_trend_slope', 'amount_acceleration', 
                    'amount_first_vs_last', 'amount_increase_ratio', 'amount_change_anomaly']]

tx_time_concentration_profile — 计算交易时间集中度画像。洗钱交易时间分布更集中（尖峰分布），集中在特定时段操作； 
def feature_tx_time_concentration_profile(open_df, tx_df, close_df):
    """
    交易时间集中度画像：计算每个钱包交易时间集中在几个核心时段，
    以及交易时间分布的峰度（kurtosis）。洗钱交易时间分布更集中（尖峰），
    正常交易更分散。
    """
    import pandas as pd
    import numpy as np
    from scipy import stats
    
    tx = tx_df.copy()
    
    # 计算每个钱包的交易时间分布统计
    def calc_time_profile(group):
        hours = group['hour'].values
        if len(hours) < 3:
            return pd.Series({
                'hour_mode': 0,
                'hour_mode_ratio': 0.0,
                'hour_kurtosis': 0.0,
                'hour_skew': 0.0,
                'hour_iqr': 0.0,
                'peak_hour_count': 0
            })
        
        # 最频繁交易小时
        from collections import Counter
        counts = Counter(hours)
        mode_hour = counts.most_common(1)[0][0]
        mode_ratio = counts.most_common(1)[0][1] / len(hours)
        
        # 峰度（分布尖峰程度）
        kurt = stats.kurtosis(hours)
        skew = stats.skew(hours)
        
        # 四分位距
        q75, q25 = np.percentile(hours, [75, 25])
        iqr = q75 - q25
        
        # 峰值小时数（交易占比>15%的小时数）
        peak_hours = sum(1 for h, c in counts.items() if c / len(hours) > 0.15)
        
        return pd.Series({
            'hour_mode': mode_hour,
            'hour_mode_ratio': mode_ratio,
            'hour_kurtosis': kurt,
            'hour_skew': skew,
            'hour_iqr': iqr,
            'peak_hour_count': peak_hours
        })
    
    src_profile = tx.groupby('src').apply(calc_time_profile).reset_index()
    src_profile.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    # 目标钱包维度
    dst_profile = tx.groupby('dst')['hour'].agg(
        lambda x: stats.kurtosis(x) if len(x) >= 3 else 0
    ).reset_index()
    dst_profile.columns = ['wallet_id', 'dst_hour_kurtosis']
    
    result = src_profile.merge(dst_profile, on='wallet_id', how='outer')
    result = result.fillna(0)
    
    # 异常分数：高峰值+高mode占比+少peak小时 = 时间集中（异常）
    result['time_concentration_score'] = result['hour_mode_ratio'] * (1 - result['peak_hour_count'] / 12)
    
    return result[['wallet_id', 'hour_mode', 'hour_mode_ratio', 'hour_kurtosis', 
                    'hour_skew', 'hour_iqr', 'peak_hour_count', 'dst_hour_kurtosis', 'time_concentration_score']]

result = feature_tx_time_concentration_profile(open_df, tx_df, close_df)

def feature_day_of_week_amount_pattern(open_df, tx_df, close_df):

    # 提取星期几

            return pd.Series({'weekday_amount_cv': 0.0, 'weekday_active_days': len(amounts)})
        mean = np.mean(amounts)
        std = np.std(amounts)

    workday_amount.columns = ['wallet_id', 'workday_avg_amount']

    # 异常分数：低变异系数（每天金额均匀）+ 周末周中金额比接近1 = 不受星期影响（异常）

    return result[['wallet_id', 'weekday_amount_cv', 'weekday_count_cv', 'weekday_active_days',

tx_amount_ratio_analysis — 计算交易金额比例分析。洗钱交易中转出金额与转入金额存在固定比例关系（佣金扣减1% 
def feature_tx_amount_ratio_analysis(open_df, tx_df, close_df):
    """
    交易金额比例分析：洗钱交易中转出金额与转入金额存在固定比例关系
    （佣金扣减），计算连续交易的金额比例规律性。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    def calc_amount_ratio(group):
        amounts = group['amount'].values
        if len(amounts) < 4:
            return pd.Series({
                'amount_ratio_cv': 0.0,
                'amount_ratio_entropy': 0.0,
                'amount_round_ratio': 0.0,
                'amount_gap_to_limit': 999999.0
            })
        
        # 相邻金额比
        ratios = [amounts[i+1] / (amounts[i] + 0.01) for i in range(len(amounts)-1)]
        
        # 比例变异系数（越小越规律）
        ratio_mean = np.mean(ratios)
        ratio_std = np.std(ratios)
        ratio_cv = ratio_std / (ratio_mean + 0.01)
        
        # 比例分布熵（将比例分桶）
        ratio_buckets = [int(r * 10) for r in ratios]  # 0.1精度
        from collections import Counter
        counts = Counter(ratio_buckets)
        total = len(ratio_buckets)
        probs = [c/total for c in counts.values()]
        entropy = -sum(p * np.log2(p + 1e-10) for p in probs)
        
        # 金额接近常见限额的比例
        limits = [5000, 10000, 20000, 50000]
        near_limit = 0
        for a in amounts:
            for lim in limits:
                if abs(a - lim) / lim < 0.05:
                    near_limit += 1
                    break
        round_ratio = near_limit / len(amounts)
        
        # 与最近限额的最小距离
        min_gap = 999999
        for a in amounts:
            for lim in limits:
                gap = abs(a - lim)

tx_interval_autocorrelation — 计算交易间隔序列的自相关性。洗钱操作中交易间隔呈现规律性模式（如固定间隔转出）， 
def feature_tx_interval_autocorrelation(open_df, tx_df, close_df):
    """
    交易间隔自相关性：洗钱操作中交易间隔呈现规律性模式（如固定间隔转出），
    导致间隔序列自相关高。正常交易间隔随机，自相关接近0。
    计算间隔序列的自相关系数。
    """
    import pandas as pd
    import numpy as np
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    def calc_interval_acf(group):
        intervals = group['interval'].values
        if len(intervals) < 5:
            return pd.Series({
                'interval_acf_lag1': 0.0,
                'interval_acf_lag2': 0.0,
                'interval_acf_lag3': 0.0,
                'interval_mean_abs_acf': 0.0,
                'interval_ljung_box': 0.0
            })
        
        # 计算自相关
        n = len(intervals)
        mean_iv = np.mean(intervals)
        var_iv = np.var(intervals)
        
        if var_iv < 1e-10:
            return pd.Series({
                'interval_acf_lag1': 0.0,
                'interval_acf_lag2': 0.0,
                'interval_acf_lag3': 0.0,
                'interval_mean_abs_acf': 0.0,
                'interval_ljung_box': 0.0
            })
        
        # lag-1自相关
        c0 = np.sum((intervals - mean_iv) ** 2) / n
        c1 = np.sum((intervals[:-1] - mean_iv) * (intervals[1:] - mean_iv)) / n
        acf1 = c1 / c0 if c0 > 0 else 0
        
        # lag-2自相关
        c2 = np.sum((intervals[:-2] - mean_iv) * (intervals[2:] - mean_iv)) / n
        acf2 = c2 / c0 if c0 > 0 else 0
        
        # lag-3自相关
        c3 = np.sum((intervals[:-3] - mean_iv) * (intervals[3:] - mean_iv)) / n
        acf3 = c3 / c0 if c0 > 0 else 0
        
        # 平均绝对自相关
        mean_abs_acf = (abs(acf1) + abs(acf2) + abs(acf3)) / 3
        
        # Ljung-Box统计量（简化）
        lb = n * (n + 2) * (acf1**2 / (n - 1) + acf2**2 / (n - 2) + acf3**2 / (n - 3))
        
        return pd.Series({
            'interval_acf_lag1': acf1,
            'interval_acf_lag2': acf2,
            'interval_acf_lag3': acf3,
            'interval_mean_abs_acf': mean_abs_acf,
            'interval_ljung_box': lb
        })
    
    acf_features = tx_sorted.groupby('src').apply(calc_interval_acf).reset_index()
    acf_features.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result = acf_features.fillna(0)
    
    # 异常分数：高自相关 = 间隔有规律（异常）
    result['interval_regularity_score'] = result['interval_mean_abs_acf']
    
    return result[['wallet_id', 'interval_acf_lag1', 'interval_acf_lag2', 'interval_acf_lag3',
                    'interval_mean_abs_acf', 'interval_ljung_box', 'interval_regularity_score']]

tx_time_gap_clustering — 计算交易间隔聚类特征。洗钱交易间隔呈双峰分布（短间隔的连续操作+长间隔的等待期） 
def feature_tx_time_gap_clustering(open_df, tx_df, close_df):
    """
    交易时间间隔聚类特征：洗钱交易间隔呈现双峰分布（极短间隔+较长间隔），
    正常交易间隔分布更连续。计算间隔分布的聚类特征。
    """
    import pandas as pd
    import numpy as np
    from collections import Counter
    
    tx = tx_df.copy()
    tx['timestamp'] = pd.to_datetime(tx['timestamp'])
    tx_sorted = tx.sort_values(['src', 'timestamp'])
    
    def calc_gap_clustering(group):
        intervals = group['interval'].values
        if len(intervals) < 5:
            return pd.Series({
                'gap_bimodality': 0.0,
                'short_gap_ratio': 0.0,
                'medium_gap_ratio': 0.0,
                'long_gap_ratio': 0.0,
                'gap_90p_vs_10p': 0.0
            })
        
        # 将间隔分为3类：短(<60s)、中(60-600s)、长(>600s)
        short = np.sum(intervals < 60) / len(intervals)
        medium = np.sum((intervals >= 60) & (intervals < 600)) / len(intervals)
        long_ = np.sum(intervals >= 600) / len(intervals)
        
        # 双峰性指标（短+长占比高，中占比低 = 双峰）
        bimodality = short + long_ - medium
        
        # 90分位/10分位比
        q90, q10 = np.percentile(intervals, [90, 10])
        ratio = q90 / (q10 + 1)
        
        return pd.Series({
            'gap_bimodality': max(bimodality, 0),
            'short_gap_ratio': short,
            'medium_gap_ratio': medium,
            'long_gap_ratio': long_,
            'gap_90p_vs_10p': ratio
        })
    
    gap_features = tx_sorted.groupby('src').apply(calc_gap_clustering).reset_index()
    gap_features.rename(columns={'src': 'wallet_id'}, inplace=True)
    
    result = gap_features.fillna(0)
    
    return result[['wallet_id', 'gap_bimodality', 'short_gap_ratio', 
                    'medium_gap_ratio', 'long_gap_ratio', 'gap_90p_vs_10p']]
